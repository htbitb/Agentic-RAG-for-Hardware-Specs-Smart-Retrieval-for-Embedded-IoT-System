#include "stm32g4xx.h"

void usart1_init(void)
{
    /* 1. Enable GPIOA and USART1 peripheral clocks */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB2ENR |= RCC_APB2ENR_USART1EN;

    /* 2. Configure PA9 (TX) and PA10 (RX) for Alternate Function mode (AF7) */
    // Set PA9 and PA10 to Alternate Function (MODER = 0b10)
    GPIOA->MODER &= ~(GPIO_MODER_MODE9_Msk | GPIO_MODER_MODE10_Msk);
    GPIOA->MODER |= (2UL << GPIO_MODER_MODE9_Pos) | (2UL << GPIO_MODER_MODE10_Pos);

    // Set PA9 to high speed
    GPIOA->OSPEEDR |= (2UL << GPIO_OSPEEDR_OSPEED9_Pos);

    // Enable pull-up on PA10 (RX) to keep line idle high when disconnected
    GPIOA->PUPDR &= ~GPIO_PUPDR_PUPD10_Msk;
    GPIOA->PUPDR |= (1UL << GPIO_PUPDR_PUPD10_Pos);

    // Map PA9 and PA10 to AF7 (USART1) in AFRH (pins 9 and 10)
    GPIOA->AFR[1] &= ~((0xFUL << GPIO_AFRH_AFSEL9_Pos) | (0xFUL << GPIO_AFRH_AFSEL10_Pos));
    GPIOA->AFR[1] |=  (7UL   << GPIO_AFRH_AFSEL9_Pos) | (7UL   << GPIO_AFRH_AFSEL10_Pos);

    /* 3. Configure USART1 */
    // Ensure USART1 is disabled before configuration
    USART1->CR1 = 0;

    // Default after reset is 1 stop bit (CR2 STOP = 00), 8 data bits (M0=0, M1=0),
    // no parity (PCE=0), and oversampling by 16 (OVER8=0).
    USART1->CR2 = 0;
    USART1->CR3 = 0;

    // USART1 clock source defaults to PCLK2 (170 MHz) in RCC->CCIPR (USART1SEL = 0b00).
    // BRR = f_CK / Baud = 170000000 / 115200 ≈ 1475.694 -> 1476
    // Actual baud = 170000000 / 1476 = 115176.15 baud (error = -0.021%, well within ±2%)
    USART1->BRR = 1476UL;

    // Enable Transmitter (TE), Receiver (RE), and USART (UE)
    USART1->CR1 = USART_CR1_TE | USART_CR1_RE | USART_CR1_UE;
}
```

### Explanation

1. **Clocks**: The peripheral clocks for `GPIOA` (on AHB2) and `USART1` (on APB2) are enabled. By default, `RCC->CCIPR` selects PCLK2 (170 MHz) as the USART1 clock source.
2. **GPIO Configuration**: Pins `PA9` and `PA10` are configured as alternate function mode (`MODER = 0b10`) and routed to AF7 (`USART1_TX` and `USART1_RX`) via `AFR[1]`.
3. **Baud Rate**: For 16× oversampling (`OVER8 = 0`), the baud-rate register directly holds the unsigned divider value: $\text{BRR} = 170\text{ MHz} / 115200 \approx 1476$. This yields an actual baud rate of 115176 baud ($-0.021\%$ error).
4. **USART Control**: Frame format parameters (8 data bits, no parity, 1 stop bit) default to reset values. The baud rate is written to `BRR` while the USART is disabled (`UE = 0`), after which transmitter (`TE`), receiver (`RE`), and the peripheral (`UE`) are activated in `CR1`.

#include "stm32g4xx.h"

void i2c2_init(void)
{
    /* 1. Enable GPIOA and I2C2 peripheral clocks */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_I2C2EN;

    /* 2. Configure PA8 (SDA) and PA9 (SCL) */
    // Alternate function mode (MODER = 0b10)
    GPIOA->MODER &= ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk);
    GPIOA->MODER |=  (2UL << GPIO_MODER_MODE8_Pos) | (2UL << GPIO_MODER_MODE9_Pos);

    // Open-drain output type (OTYPER = 1)
    GPIOA->OTYPER |= GPIO_OTYPER_OT8 | GPIO_OTYPER_OT9;

    // High speed
    GPIOA->OSPEEDR |= (2UL << GPIO_OSPEEDR_OSPEED8_Pos) | (2UL << GPIO_OSPEEDR_OSPEED9_Pos);

    // Internal pull-up enabled (optional good practice with open-drain)
    GPIOA->PUPDR &= ~(GPIO_PUPDR_PUPD8_Msk | GPIO_PUPDR_PUPD9_Msk);
    GPIOA->PUPDR |=  (1UL << GPIO_PUPDR_PUPD8_Pos) | (1UL << GPIO_PUPDR_PUPD9_Pos);

    // Alternate function AF4 for PA8 (I2C2_SDA) and PA9 (I2C2_SCL)
    GPIOA->AFR[1] &= ~((0xFUL << GPIO_AFRH_AFSEL8_Pos) | (0xFUL << GPIO_AFRH_AFSEL9_Pos));
    GPIOA->AFR[1] |=  (4UL   << GPIO_AFRH_AFSEL8_Pos) | (4UL   << GPIO_AFRH_AFSEL9_Pos);

    /* 3. Configure I2C2 */
    // Ensure peripheral is disabled (PE = 0) before setting TIMINGR
    I2C2->CR1 &= ~I2C_CR1_PE;

    /*
     * TIMINGR calculation for 400 kHz Fast Mode:
     * Kernel clock = PCLK1 = 170 MHz (reset default, RCC_CCIPR I2C2SEL = 0b00).
     *
     * PRESC = 16 - 1 = 15 (0xF) -> t_PRESC = (15 + 1) / 170 MHz = 16 / 170 MHz ≈ 94.12 ns
     *
     * Minimum requirements (RM0440 Table 358):
     *   t_LOW  >= 1.3 µs -> SCLL = 15 -> t_LOW  = (15 + 1) * 94.12 ns = 1.506 µs (>= 1.3 µs)
     *   t_HIGH >= 0.6 µs -> SCLH = 8  -> t_HIGH = (8  + 1) * 94.12 ns = 0.847 µs (>= 0.6 µs)
     *   Total nominal period = (16 + 9) * 94.12 ns = 2.353 µs; with SCL rise time (~150 ns),
     *   period ≈ 2.5 µs -> ~400 kHz.
     *
     * Data setup/hold timings:
     *   SCLDEL = 3 -> t_SCLDEL = (3 + 1) * 94.12 ns = 376.5 ns (t_SU:DAT min = 100 ns)
     *   SDADEL = 2 -> t_SDADEL = 2 * 94.12 ns = 188.2 ns (t_HD:DAT min = 0 ns, max = 900 ns)
     *
     * TIMINGR = (PRESC << 28) | (SCLDEL << 20) | (SDADEL << 16) | (SCLH << 8) | SCLL
     */
    I2C2->TIMINGR = (0xFUL << I2C_TIMINGR_PRESC_Pos)  |
                    (0x3UL << I2C_TIMINGR_SCLDEL_Pos) |
                    (0x2UL << I2C_TIMINGR_SDADEL_Pos) |
                    (0x08UL << I2C_TIMINGR_SCLH_Pos)  |
                    (0x0FUL << I2C_TIMINGR_SCLL_Pos);

    // Enable I2C2 peripheral
    I2C2->CR1 |= I2C_CR1_PE;
}
```

### Explanation

1. **Clocks**: Enabled `RCC_AHB2ENR_GPIOAEN` and `RCC_APB1ENR1_I2C2EN`. `RCC->CCIPR` defaults to `0b00` for `I2C2SEL`, keeping the kernel clock at PCLK1 ($170\text{ MHz}$).
2. **GPIO Configuration**: Configured pins `PA8` (SDA) and `PA9` (SCL) to Alternate Function mode with Open-Drain output (`OTYPER = 1`) and mapped them to AF4 (`I2C2_SDA` and `I2C2_SCL`) in `AFR[1]`.
3. **Timing Register (`TIMINGR`)**:
   - `PRESC = 15` divides the $170\text{ MHz}$ clock by 16, resulting in $t_{\text{PRESC}} \approx 94.12\text{ ns}$.
   - `SCLL = 15` gives $t_{\text{LOW}} = (15 + 1) \times 94.12\text{ ns} \approx 1.51\ \mu\text{s}$, satisfying the $\ge 1.3\ \mu\text{s}$ fast-mode limit [RM0440 p.1017].
   - `SCLH = 8` gives $t_{\text{HIGH}} = (8 + 1) \times 94.12\text{ ns} \approx 0.85\ \mu\text{s}$, satisfying the $\ge 0.6\ \mu\text{s}$ fast-mode limit [RM0440 p.1017].
   - `SCLDEL = 3` and `SDADEL = 2` meet the required setup ($\ge 100\text{ ns}$) and hold times.
4. **Enabling**: `I2C_TIMINGR` is written while `PE = 0`, then `I2C_CR1_PE` is asserted.

int main(void)
{
    /* -------------------------------------------------------------------------
     * 1. Enable Peripheral Clocks
     * -------------------------------------------------------------------------
     * GPIOA : AHB2
     * USART1: APB2 (kernel clock defaults to PCLK2 = 170 MHz via RCC->CCIPR)
     * I2C2  : APB1 (kernel clock defaults to PCLK1 = 170 MHz via RCC->CCIPR)
     */

    RCC->APB2ENR  |= RCC_APB2ENR_USART1EN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_I2C2EN;

    /* -------------------------------------------------------------------------
     * 2. GPIO Configuration
     * -------------------------------------------------------------------------
     * Pin assignments:
     *   PA8 : I2C2_SDA  (AF4, Open-Drain, Pull-up, High Speed)
     *   PA10: USART1_RX (AF7, Input/AF,   Pull-up, High Speed)
     *   PA9 : Pin conflict in requested design (USART1_TX uses AF7, I2C2_SCL uses AF4).
     *         Configured below for AF4 (I2C2_SCL, Open-Drain) as requested in Message 2.
     */

    GPIOA->MODER &= ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk | GPIO_MODER_MODE10_Msk);
    GPIOA->MODER |=  (2UL << GPIO_MODER_MODE8_Pos)
                  |  (2UL << GPIO_MODER_MODE9_Pos)
                  |  (2UL << GPIO_MODER_MODE10_Pos);

    // PA8 and PA9: Open-drain (I2C standard requirements)
    GPIOA->OTYPER |= GPIO_OTYPER_OT8 | GPIO_OTYPER_OT9;

    // High speed for PA8, PA9, PA10
    GPIOA->OSPEEDR |= (2UL << GPIO_OSPEEDR_OSPEED8_Pos)
                   |  (2UL << GPIO_OSPEEDR_OSPEED9_Pos)
                   |  (2UL << GPIO_OSPEEDR_OSPEED10_Pos);

    // Pull-up for PA8 (SDA), PA9 (SCL), and PA10 (RX)
    GPIOA->PUPDR &= ~(GPIO_PUPDR_PUPD8_Msk | GPIO_PUPDR_PUPD9_Msk | GPIO_PUPDR_PUPD10_Msk);
    GPIOA->PUPDR |=  (1UL << GPIO_PUPDR_PUPD8_Pos)
                 |  (1UL << GPIO_PUPDR_PUPD9_Pos)
                 |  (1UL << GPIO_PUPDR_PUPD10_Pos);

    // Alternate function routing in AFRH (pins 8, 9, 10):
    // PA8  -> AF4 (I2C2_SDA)
    // PA9  -> AF4 (I2C2_SCL)  [Note: AF7 required if used as USART1_TX]
    // PA10 -> AF7 (USART1_RX)

                       (0xFUL << GPIO_AFRH_AFSEL9_Pos)  |
                       (0xFUL << GPIO_AFRH_AFSEL10_Pos));

    GPIOA->AFR[1] |=  (4UL   << GPIO_AFRH_AFSEL8_Pos)   |
                      (4UL   << GPIO_AFRH_AFSEL9_Pos)   |
                      (7UL   << GPIO_AFRH_AFSEL10_Pos);

    /* -------------------------------------------------------------------------
     * 3. USART1 Peripheral Configuration (115200 8N1, 16x oversampling)
     * -------------------------------------------------------------------------
     */
    USART1->CR1 = 0; // Peripheral disabled during configuration
    USART1->CR2 = 0; // 1 stop bit
    USART1->CR3 = 0;

    // BRR = 170 MHz / 115200 ≈ 1476 (-0.021% baud error)
    USART1->BRR = 1476UL;

    // Enable Transmitter, Receiver, and USART peripheral
    USART1->CR1 = USART_CR1_TE | USART_CR1_RE | USART_CR1_UE;

     * 4. I2C2 Peripheral Configuration (400 kHz Fast Mode)
     * -------------------------------------------------------------------------

                    (0x2UL  << I2C_TIMINGR_SDADEL_Pos) |
                    (0x08UL << I2C_TIMINGR_SCLH_Pos)   |
                    (0x0FUL << I2C_TIMINGR_SCLL_Pos);

    // Enable I2C2 peripheral
    I2C2->CR1 |= I2C_CR1_PE;

    /* -------------------------------------------------------------------------
     * 5. Infinite Loop
     * -------------------------------------------------------------------------
     */

    {
    }

    return 0;

### Explanation

1. **Clocks**: The peripheral bus clocks for `GPIOA` (AHB2), `USART1` (APB2), and `I2C2` (APB1) are enabled.
2. **Pin Assignment Conflict**:
   - Message 1 requested `PA9` for `USART1_TX` (AF7).
   - Message 2 requested `PA9` for `I2C2_SCL` (AF4).
   - Because a physical pin can map to only one alternate function at a time, `PA9` is configured for AF4/Open-Drain (I2C2) in this combined initialization. To make both peripherals concurrently functional, one of them must be remapped to an alternate pin (e.g., `USART1_TX` to `PB6` or `PC4`, or `I2C2_SCL` to `PC4`).
3. **USART1**: Configured for 115200 baud (BRR = 1476 at 170 MHz), 8 data bits, 1 stop bit, no parity, and 16× oversampling with `TE`, `RE`, and `UE` asserted.
4. **I2C2**: `I2C_TIMINGR` is written while `PE = 0` to satisfy Fast Mode ($400\text{ kHz}$) constraints ($t_{\text{LOW}} \ge 1.3\ \mu\text{s}$, $t_{\text{HIGH}} \ge 0.6\ \mu\text{s}$ [RM0440 p.1017]), then enabled via `I2C_CR1_PE`.
