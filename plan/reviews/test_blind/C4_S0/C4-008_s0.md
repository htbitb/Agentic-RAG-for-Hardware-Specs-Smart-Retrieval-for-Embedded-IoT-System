Write bare-metal C that accesses registers the CMSIS way (e.g. `TIM1->PSC = ...;`, `RCC->APB2ENR |= RCC_APB2ENR_TIM1EN;`) using the register and bit names of ST's `stm32g4xx.h` device header. Message 1: configure USART1 for 115200 baud with TX on pin PA9 and RX on pin PA10. ### Assistant answer 0

```c
#include "stm32g4xx.h"

void usart1_init(void)
{
    /* 1. Enable peripheral clocks for GPIOA and USART1 */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB2ENR |= RCC_APB2ENR_USART1EN;

    /* 2. Configure PA9 (TX) and PA10 (RX) to Alternate Function mode (AF7) */
    GPIOA->MODER &= ~(GPIO_MODER_MODE9_Msk | GPIO_MODER_MODE10_Msk);
    GPIOA->MODER |= (2U << GPIO_MODER_MODE9_Pos) | (2U << GPIO_MODER_MODE10_Pos);

    /* AF7 (USART1_TX/RX) mapping in AFRH (AFR[1]) */
    GPIOA->AFR[1] &= ~(GPIO_AFRH_AFSEL9_Msk | GPIO_AFRH_AFSEL10_Msk);
    GPIOA->AFR[1] |= (7U << GPIO_AFRH_AFSEL9_Pos) | (7U << GPIO_AFRH_AFSEL10_Pos);

    /* Set PA9 to high speed */
    GPIOA->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED9_Pos);

    /* 3. Configure frame format: 8 data bits, no parity, 1 stop bit, oversampling by 16 */
    USART1->CR1 = 0; /* Clears M0, M1 (8 data bits), PCE (no parity), OVER8 (oversampling by 16) */
    USART1->CR2 &= ~USART_CR2_STOP_Msk; /* 00 = 1 stop bit */

    /* 
     * 4. Configure Baud Rate before enabling USART
     * USART1 clock source defaults to PCLK2 = 170 MHz (RCC->CCIPR USART1SEL = 00).
     * With OVER8 = 0: USARTDIV = fCK / baud = 170000000 / 115200 = 1475.694...
     * BRR = 1476 -> Actual baud = 170000000 / 1476 = 115176.15 baud (error = -0.02%, well within ±2%)
     */
    USART1->BRR = 1476U;

    /* 5. Enable transmitter (TE), receiver (RE), and the USART module (UE) */
    USART1->CR1 |= (USART_CR1_TE | USART_CR1_RE);
    USART1->CR1 |= USART_CR1_UE;
}
```

### Explanation

* **Pin Mapping**: On STM32G474, pins PA9 and PA10 multiplex `USART1_TX` and `USART1_RX` on Alternate Function 7 (`AF7`) [DS12288 p.74].
* **Clock & Baud Rate**: At reset, USART1 selects `PCLK2` (170 MHz) as its clock source (`RCC_CCIPR_USART1SEL = 0`). With 16× oversampling (`OVER8 = 0`), the baud-rate divider is directly written to `BRR` as `170000000 / 115200 ≈ 1476`, yielding an error of only $-0.02\%$.
* **Sequencing**: Frame formatting and baud rate are programmed prior to setting `USART_CR1_UE` to meet the peripheral configuration requirements [RM0440 p.1384].

```c
#include "stm32g4xx.h"

void i2c2_init(void)
{
    /* 1. Enable peripheral clocks for GPIOA and I2C2 */
    RCC->AHB2ENR   |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB1ENR1  |= RCC_APB1ENR1_I2C2EN;

    /* 2. Configure PA8 (SDA) and PA9 (SCL) */
    /* Alternate Function mode (10b) */
    GPIOA->MODER &= ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk);
    GPIOA->MODER |= (2U << GPIO_MODER_MODE8_Pos) | (2U << GPIO_MODER_MODE9_Pos);

    /* Open-drain output type */
    GPIOA->OTYPER |= (GPIO_OTYPER_OT8 | GPIO_OTYPER_OT9);

    /* High speed */
    GPIOA->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED8_Pos) | (3U << GPIO_OSPEEDR_OSPEED9_Pos);

    /* AF4 (I2C2_SDA / I2C2_SCL) in AFRH (AFR[1]) */
    GPIOA->AFR[1] &= ~(GPIO_AFRH_AFSEL8_Msk | GPIO_AFRH_AFSEL9_Msk);
    GPIOA->AFR[1] |= (4U << GPIO_AFRH_AFSEL8_Pos) | (4U << GPIO_AFRH_AFSEL9_Pos);

    /* 
     * 3. Configure timing while I2C2 is disabled (PE = 0)
     * Kernel clock = PCLK1 = 170 MHz (reset default in RCC->CCIPR).
     * t_I2CCLK = 1 / 170 MHz ≈ 5.882 ns.
     *
     * Timing calculation for 400 kHz Fast-mode (T = 2500 ns):
     * - PRESC = 4  -> t_PRESC = (4 + 1) * 5.882 ns = 29.412 ns
     * - SCLL  = 53 -> t_LOW   = (53 + 1) * 29.412 ns = 1588.2 ns (>= 1300 ns, RM0440 Table 358)
     * - SCLH  = 30 -> t_HIGH  = (30 + 1) * 29.412 ns = 911.8 ns  (>= 600 ns,  RM0440 Table 358)
     * - Total SCL period = (54 + 31) * 29.412 ns = 2500 ns -> f_SCL = 400.0 kHz
     * - SDADEL = 1 -> t_SDADEL = 1 * 29.412 ns = 29.4 ns (data hold)
     * - SCLDEL = 3 -> t_SCLDEL = (3 + 1) * 29.412 ns = 117.6 ns (>= 100 ns setup time)
     */
    I2C2->CR1 &= ~I2C_CR1_PE;

    I2C2->TIMINGR = (4U  << I2C_TIMINGR_PRESC_Pos)  |
                    (3U  << I2C_TIMINGR_SCLDEL_Pos) |
                    (1U  << I2C_TIMINGR_SDADEL_Pos) |
                    (30U << I2C_TIMINGR_SCLH_Pos)   |
                    (53U << I2C_TIMINGR_SCLL_Pos);

    /* 4. Enable I2C2 */
    I2C2->CR1 |= I2C_CR1_PE;
}
```

### Explanation

* **Pin Mapping & Output Type**: `PA8` (SDA) and `PA9` (SCL) multiplex with `I2C2` via Alternate Function 4 (`AF4`) [DS12288 p.74]. Both pins are configured as open-drain via `GPIOx_OTYPER` as required by the $I^2C$ specification.
* **Kernel Clock**: At reset, `RCC->CCIPR` bits `I2C2SEL[1:0] = 00b`, selecting `PCLK1 = 170 MHz` [RM0440 p.254].
* **Timing Calculation**: Setting `PRESC = 4` scales the 170 MHz clock down to 34 MHz ($t_{PRESC} \approx 29.412\text{ ns}$). Dividing 400 kHz into $54$ prescaled cycles low ($1.588\ \mu\text{s} \ge 1.3\ \mu\text{s}$) and $31$ prescaled cycles high ($0.912\ \mu\text{s} \ge 0.6\ \mu\text{s}$) gives exactly $85\text{ ticks} = 2.5\ \mu\text{s}$ (400.0 kHz), satisfying RM0440 Table 358 [RM0440 p.1311].
* **Sequencing**: `I2C_TIMINGR` is written while `PE = 0` before setting `I2C_CR1_PE` [RM0440 p.1337].

```c
#include "stm32g4xx.h"

int main(void)
{
    /* 1. Enable peripheral clocks */
    RCC->AHB2ENR  |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_I2C2EN;
    RCC->APB2ENR  |= RCC_APB2ENR_USART1EN;

    /* 
     * 2. Configure GPIOA pins:
     * PA8  -> I2C2_SDA   (AF4, Open-Drain, High-Speed)
     * PA9  -> Pin conflict: Requested for USART1_TX (AF7) and I2C2_SCL (AF4)
     *         (Configured below as AF4 / Open-Drain per Message 2)
     * PA10 -> USART1_RX  (AF7, Push-Pull, High-Speed)
     */
    GPIOA->MODER &= ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk | GPIO_MODER_MODE10_Msk);
    GPIOA->MODER |= (2U << GPIO_MODER_MODE8_Pos)  |
                    (2U << GPIO_MODER_MODE9_Pos)  |
                    (2U << GPIO_MODER_MODE10_Pos);

    /* Open-drain for I2C pins PA8 and PA9 */
    GPIOA->OTYPER |= (GPIO_OTYPER_OT8 | GPIO_OTYPER_OT9);

    /* High speed on all communication pins */
    GPIOA->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED8_Pos)  |
                      (3U << GPIO_OSPEEDR_OSPEED9_Pos)  |
                      (3U << GPIO_OSPEEDR_OSPEED10_Pos);

    /* Alternate functions: PA8 = AF4, PA9 = AF4 (I2C2_SCL), PA10 = AF7 (USART1_RX) */
    GPIOA->AFR[1] &= ~(GPIO_AFRH_AFSEL8_Msk | GPIO_AFRH_AFSEL9_Msk | GPIO_AFRH_AFSEL10_Msk);
    GPIOA->AFR[1] |= (4U << GPIO_AFRH_AFSEL8_Pos)  |
                     (4U << GPIO_AFRH_AFSEL9_Pos)  |
                     (7U << GPIO_AFRH_AFSEL10_Pos);

    /* 
     * 3. Configure USART1 (115200 baud, 8N1, 16x oversampling)
     * Clock = PCLK2 = 170 MHz. BRR = 170000000 / 115200 = 1476 (-0.02% error).
     */
    USART1->CR1 = 0;
    USART1->CR2 &= ~USART_CR2_STOP_Msk;
    USART1->BRR = 1476U;
    USART1->CR1 |= (USART_CR1_TE | USART_CR1_RE);
    USART1->CR1 |= USART_CR1_UE;

    /* 
     * 4. Configure I2C2 (Fast mode 400 kHz)
     * Clock = PCLK1 = 170 MHz. PRESC = 4, SCLL = 53, SCLH = 30, SDADEL = 1, SCLDEL = 3.
     */
    I2C2->CR1 &= ~I2C_CR1_PE;
    I2C2->TIMINGR = (4U  << I2C_TIMINGR_PRESC_Pos)  |
                    (3U  << I2C_TIMINGR_SCLDEL_Pos) |
                    (1U  << I2C_TIMINGR_SDADEL_Pos) |
                    (30U << I2C_TIMINGR_SCLH_Pos)   |
                    (53U << I2C_TIMINGR_SCLL_Pos);
    I2C2->CR1 |= I2C_CR1_PE;

    /* Loop forever */
    while (1)
    {
    }
}
```

### Explanation

* **Peripheral Initialization**: Both peripheral clocks (`USART1` on APB2, `I2C2` on APB1, and `GPIOA` on AHB2) are enabled. The baud rate (1476) and I2C timings are configured before setting `UE` and `PE` respectively [RM0440 p.1337, p.1384].
* **Pin Conflict on PA9**: Note that **PA9** was specified in Message 1 for `USART1_TX` (AF7) and in Message 2 for `I2C2_SCL` (AF4). Because a physical pin can map to only one Alternate Function at a time, assigning PA9 to `I2C2_SCL` prevents `USART1_TX` from outputting on that pin. If concurrent operation is needed, one of the two signals should be remapped to an alternate pin (e.g., `USART1_TX` on `PB6` [AF7] or `PC4` [AF7]; or `I2C2_SCL` on `PC4` [AF8]) [DS12288 p.74, p.76].
