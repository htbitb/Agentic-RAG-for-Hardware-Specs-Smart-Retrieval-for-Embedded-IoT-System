```c
#include "stm32g4xx.h"

void usart2_init(void)
{
    /* 1. Enable peripheral clocks for GPIOA and USART2 */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_USART2EN;

    /* 2. Configure PA2 (TX) and PA3 (RX) for Alternate Function 7 (USART2) */
    /* Set mode to Alternate Function (10b) */
    GPIOA->MODER &= ~(GPIO_MODER_MODE2_Msk | GPIO_MODER_MODE3_Msk);
    GPIOA->MODER |= (2UL << GPIO_MODER_MODE2_Pos) | (2UL << GPIO_MODER_MODE3_Pos);

    /* Set AF7 on PA2 and PA3 */
    GPIOA->AFR[0] &= ~(GPIO_AFRL_AFSEL2_Msk | GPIO_AFRL_AFSEL3_Msk);
    GPIOA->AFR[0] |= (7UL << GPIO_AFRL_AFSEL2_Pos) | (7UL << GPIO_AFRL_AFSEL3_Pos);

    /* Set I/O output speed to very high for PA2 and PA3 */
    GPIOA->OSPEEDR |= (3UL << GPIO_OSPEEDR_OSPEED2_Pos) | (3UL << GPIO_OSPEEDR_OSPEED3_Pos);

    /* 3. Configure USART2 frame: 8 data bits, no parity, 1 stop bit, oversampling by 16 */
    /* CR1 reset state has M[1:0] = 00 (8 data bits), PCE = 0 (no parity), OVER8 = 0 (16x oversampling) */
    /* CR2 reset state has STOP[1:0] = 00 (1 stop bit) */
    USART2->CR1 = USART_CR1_TE | USART_CR1_RE;

    /* 4. Set baud rate: f_CK = PCLK1 = 170 MHz, OVER8 = 0
     * USARTDIV = 170000000 / 115200 = 1475.694 -> rounded = 1476 (0x05C4)
     */
    USART2->BRR = (170000000UL + (115200UL / 2UL)) / 115200UL;

    /* 5. Enable USART2 */
    USART2->CR1 |= USART_CR1_UE;
}
```

### Explanation

1. **Clock Gating**: `RCC->AHB2ENR` enables the GPIOA clock, while `RCC->APB1ENR1` enables the USART2 peripheral clock. By default, `RCC->CCIPR` selects PCLK1 (170 MHz) as the USART2 clock source.
2. **GPIO Configuration**: PA2 and PA3 are configured to alternate function mode (`MODER` = `10b`) and mapped to AF7 (USART2) via `AFR[0]` (AFRL).
3. **Control Configuration**: `USART2->CR1` has transmitter (`TE`) and receiver (`RE`) enabled with default frame settings (8 data bits, no parity, 1 stop bit via `CR2`, 16× oversampling).
4. **Baud Rate Calculation**: With `OVER8 = 0`, `BRR` holds the entire clock divider `USARTDIV`. For $170\text{ MHz} / 115200$, rounding to the nearest integer yields $1476$ (0.02% error).
5. **Enabling**: `BRR` and transmitter/receiver are set up prior to asserting the peripheral enable bit (`UE`) in `USART2->CR1`.

```c
void i2c1_init(void)
{

    /* Pull-up enabled */

     * Timing selection:
     *   PRESC = 3    -> t_PRESC = (3 + 1) * t_I2CCLK = 23.529 ns
```
