```c
#include "stm32g4xx.h"

void SystemClock_Config(void)
{
    /* 1. Enable the HSE oscillator (8 MHz external crystal) and wait until ready */
    RCC->CR |= RCC_CR_HSEON;
    while (!(RCC->CR & RCC_CR_HSERDY));

    /* 2. Enable Power Controller clock and enable Range 1 Boost Mode (required for 170 MHz) */
    RCC->APB1ENR1 |= RCC_APB1ENR1_PWREN;
    PWR->CR5 &= ~PWR_CR5_R1MODE; /* 0: Boost mode enabled (f_HCLK up to 170 MHz) */

    /* 3. Configure Flash latency: 4 wait states required for 170 MHz in Range 1 Boost Mode */
    FLASH->ACR &= ~FLASH_ACR_LATENCY;
    FLASH->ACR |= FLASH_ACR_LATENCY_4WS;
    while ((FLASH->ACR & FLASH_ACR_LATENCY) != FLASH_ACR_LATENCY_4WS);

    /* 4. Configure PLL:
     *    Source = HSE (8 MHz)
     *    PLLM   = /2 -> 4 MHz VCO input (field value = 1)
     *    PLLN   = *85 -> 340 MHz VCO output
     *    PLLR   = /2 -> 170 MHz PLLCLK (field value 00 = /2)
     *    Enable PLL "R" clock output (PLLREN)
     */
    RCC->PLLCFGR = RCC_PLLCFGR_PLLSRC_HSE
                 | (1UL << RCC_PLLCFGR_PLLM_Pos)
                 | (85UL << RCC_PLLCFGR_PLLN_Pos)
                 | RCC_PLLCFGR_PLLREN;

    /* 5. Enable PLL and wait until locked */
    RCC->CR |= RCC_CR_PLLON;
    while (!(RCC->CR & RCC_CR_PLLRDY));

    /* 6. Ensure AHB, APB1, and APB2 prescalers are set to division by 1 */
    RCC->CFGR &= ~(RCC_CFGR_HPRE | RCC_CFGR_PPRE1 | RCC_CFGR_PPRE2);

    /* 7. Switch system clock (SYSCLK) to PLL and wait for switch status */
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW) | RCC_CFGR_SW_PLL;
    while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL);
}
```

### Explanation

1. **HSE**: The 8 MHz external crystal is enabled via `RCC_CR_HSEON` and polled via `RCC_CR_HSERDY`.
2. **Voltage Scaling**: At reset, the STM32G4 core operates in normal Range 1 (maximum 150 MHz). To reach 170 MHz, clearing `PWR_CR5_R1MODE` activates **Range 1 Boost Mode**.
3. **Flash Wait States**: At 170 MHz in Boost Mode, the internal flash requires 4 wait states (`FLASH_ACR_LATENCY_4WS`). The latency is set and confirmed before increasing the clock speed.
4. **PLL Configuration**: 
   $$\text{VCO}_{\text{IN}} = \frac{8\text{ MHz}}{M = 2} = 4\text{ MHz}$$
   $$\text{VCO}_{\text{OUT}} = 4\text{ MHz} \times N(85) = 340\text{ MHz}$$
   $$\text{SYSCLK} = \frac{340\text{ MHz}}{R = 2} = 170\text{ MHz}$$
   This yields exactly 170 MHz (0.0% error).
5. **Clock Switching**: `HPRE`, `PPRE1`, and `PPRE2` are cleared to 0 (divide by 1), yielding $\text{HCLK} = \text{PCLK1} = \text{PCLK2} = 170\text{ MHz}$. SYSCLK is switched to the PLL output, and `RCC_CFGR_SWS` is polled until the switch is confirmed.

```c
#include "stm32g4xx.h"

void Clock_Config_200MHz(void)
{
    /* 1. Enable HSE and wait until ready */
    RCC->CR |= RCC_CR_HSEON;
    while (!(RCC->CR & RCC_CR_HSERDY));

    /* 2. Enable PWR clock and activate Range 1 Boost Mode */
    RCC->APB1ENR1 |= RCC_APB1ENR1_PWREN;
    PWR->CR5 &= ~PWR_CR5_R1MODE;

    /* 3. Configure Flash latency: 4 wait states (or 5 WS for overclocking beyond 170 MHz) */
    FLASH->ACR &= ~FLASH_ACR_LATENCY;
    FLASH->ACR |= (4UL << FLASH_ACR_LATENCY_Pos);
    while ((FLASH->ACR & FLASH_ACR_LATENCY) != (4UL << FLASH_ACR_LATENCY_Pos));

    /* 4. Configure PLL:
     *    Source = HSE (8 MHz)
     *    PLLM   = /2 -> 4 MHz VCO input (value = 1)
     *    PLLN   = *100 -> 400 MHz VCO output
     *    PLLR   = /2 -> 200 MHz SYSCLK (value = 00)
     */
    RCC->PLLCFGR = RCC_PLLCFGR_PLLSRC_HSE
                 | (1UL << RCC_PLLCFGR_PLLM_Pos)
                 | (100UL << RCC_PLLCFGR_PLLN_Pos)
                 | RCC_PLLCFGR_PLLREN;

    /* 5. Enable PLL and wait until locked */
    RCC->CR |= RCC_CR_PLLON;
    while (!(RCC->CR & RCC_CR_PLLRDY));

    /* 6. Prescalers: AHB = 1, APB1 = 1, APB2 = 1 */
    RCC->CFGR &= ~(RCC_CFGR_HPRE | RCC_CFGR_PPRE1 | RCC_CFGR_PPRE2);

    /* 7. Switch SYSCLK to PLL */
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW) | RCC_CFGR_SW_PLL;
    while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL);
}

void TIM1_PA8_PWM_Init(void)
{
    /* 1. Enable GPIOA and TIM1 peripheral clocks */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB2ENR |= RCC_APB2ENR_TIM1EN;

    /* 2. Configure PA8 as Alternate Function AF6 (TIM1_CH1) */
    GPIOA->MODER &= ~GPIO_MODER_MODE8;
    GPIOA->MODER |= (2UL << GPIO_MODER_MODE8_Pos);        /* Alternate function mode */
    GPIOA->AFR[1] &= ~GPIO_AFRH_AFSEL8;
    GPIOA->AFR[1] |= (6UL << GPIO_AFRH_AFSEL8_Pos);       /* AF6 */
    GPIOA->OSPEEDR |= (3UL << GPIO_OSPEEDR_OSPEED8_Pos);  /* Very high speed */

    /* 3. Configure TIM1 time-base:
     *    TIM1 clock = 200 MHz (PCLK2 = 200 MHz, prescaler = 1)
     *    Desired PWM = 100 kHz -> Period = 200 MHz / 100 kHz = 2000 counts
     *    ARR = 2000 - 1 = 1999
     *    50% duty cycle: CCR1 = 2000 / 2 = 1000
     */
    TIM1->PSC = 0;
    TIM1->ARR = 1999;
    TIM1->CCR1 = 1000;

    /* 4. Configure PWM Mode 1 on Channel 1, enable CCR1 preload */
    TIM1->CCMR1 &= ~TIM_CCMR1_OC1M;
    TIM1->CCMR1 |= (TIM_CCMR1_OC1M_1 | TIM_CCMR1_OC1M_2); /* PWM Mode 1 (0110) */
    TIM1->CCMR1 |= TIM_CCMR1_OC1PE;                       /* Output preload enable */

    /* 5. Active-high polarity (CC1P = 0) and enable CH1 output */
    TIM1->CCER &= ~TIM_CCER_CC1P;
    TIM1->CCER |= TIM_CCER_CC1E;

    /* 6. Enable Auto-reload preload (ARPE) and Main Output (MOE, required for advanced timers) */
    TIM1->CR1 |= TIM_CR1_ARPE;
    TIM1->BDTR |= TIM_BDTR_MOE;

    /* 7. Force an update event to load preload registers into shadow registers */
    TIM1->EGR = TIM_EGR_UG;

    /* 8. Enable counter */
    TIM1->CR1 |= TIM_CR1_CEN;
}
```

### Explanation

* **Overclocking to 200 MHz**: Setting PLLN to 100 produces $\text{VCO}_{\text{OUT}} = 400\text{ MHz}$, yielding $200\text{ MHz}$ at the PLLR output ($400 / 2$). Note that the rated maximum frequency in Range 1 Boost Mode is 170 MHz.
* **Pin Mapping**: On STM32G474, `TIM1_CH1` on `PA8` is multiplexed on Alternate Function 6 (`AF6`).
* **Period and Resolution**: With $f_{\text{TIM1CLK}} = 200\text{ MHz}$ and $f_{\text{PWM}} = 100\text{ kHz}$, the period is exactly 2000 counts ($ARR = 1999$). This gives a fine duty-cycle step resolution of $\frac{1}{2000} = 0.05\%$.
* **Duty Cycle**: With $ARR = 1999$, setting $CCR1 = 1000$ in PWM mode 1 (active while $\text{CNT} < \text{CCR1}$) yields an exact 50% duty cycle.
* **TIM1 Output Enable**: Because TIM1 is an advanced timer, `TIM_BDTR_MOE` (Main Output Enable) must be set in addition to `TIM_CCER_CC1E` to drive the physical output pin.

```c
    GPIOA->AFR[1] &= ~GPIO_AFRH_AFSEL8;
    GPIOA->AFR[1] |= (6UL << GPIO_AFRH_AFSEL8_Pos);

    GPIOA->OSPEEDR |= (3UL << GPIO_OSPEEDR_OSPEED8_Pos);

    TIM1->CCR1 = 1000;

    TIM1->CCMR1 &= ~TIM_CCMR1_OC1M;
    TIM1->CCMR1 |= (TIM_CCMR1_OC1M_1 | TIM_CCMR1_OC1M_2);

    TIM1->CCER &= ~TIM_CCER_CC1P;
    TIM1->CCER |= TIM_CCER_CC1E;

    /* Enable auto-reload preload and Main Output (MOE) */
    TIM1->CR1 |= TIM_CR1_ARPE;
    TIM1->BDTR |= TIM_BDTR_MOE;

    TIM1->EGR = TIM_EGR_UG;

    /* Start counter */
    TIM1->CR1 |= TIM_CR1_CEN;

    while (1)
    {
    }
```

### Explanation

* **Clock Setup**: The 8 MHz HSE crystal is multiplied to 400 MHz in the PLL ($8\text{ MHz} / 2 \times 100$) and divided by 2 to yield 200 MHz SYSCLK, HCLK, PCLK1, and PCLK2. Boost Mode (`PWR_CR5_R1MODE = 0`) and 4 Flash wait states are set to support high-speed operation.
* **Pin Multiplexing**: PA8 is placed into Alternate Function mode and pointed to `AF6` (`TIM1_CH1`) with maximum slew rate for sharp switching edges.
* **PWM Generation**: With TIM1 clocked at 200 MHz, a period of 2000 counts produces a 100 kHz repetition rate ($ARR = 1999$). A match value of $CCR1 = 1000$ in PWM Mode 1 keeps the output high for the first half of the counter cycle and low for the remainder, giving an exact 50% duty cycle without interrupts. The main output is un-gated via `TIM_BDTR_MOE`.
