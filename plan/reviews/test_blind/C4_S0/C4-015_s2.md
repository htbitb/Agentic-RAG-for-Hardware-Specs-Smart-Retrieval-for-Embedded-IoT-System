     *    M = 2  => PLL input = 8 MHz / 2 = 4 MHz (PLLM = 0001b)

void USART1_Init(void)
{
    /* 1. Enable peripheral clocks for GPIOA (AHB2) and USART1 (APB2) */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB2ENR |= RCC_APB2ENR_USART1EN;

    GPIOA->MODER |= (2U << GPIO_MODER_MODE9_Pos) | (2U << GPIO_MODER_MODE10_Pos);

    /* Output speed to High speed */

    /* AF7 configuration: PA9 -> AFSEL9 = 7, PA10 -> AFSEL10 = 7 (in AFR[1] / AFRH) */
    GPIOA->AFR[1] &= ~((0xFU << GPIO_AFRH_AFSEL9_Pos) | (0xFU << GPIO_AFRH_AFSEL10_Pos));
    GPIOA->AFR[1] |= (7U << GPIO_AFRH_AFSEL9_Pos) | (7U << GPIO_AFRH_AFSEL10_Pos);

    /* 3. Configure USART1:
     *    - Reset CR1, CR2, CR3 to default (OVER8 = 0: oversample by 16, M0/M1 = 00: 8 data bits,
     *      PCE = 0: no parity, STOP[1:0] = 00: 1 stop bit)
     */

    USART1->CR2 = 0U;
    USART1->CR3 = 0U;

    /* 4. Set baud rate to 115200:
     *    USART1 clock source by default (CCIPR USART1SEL = 00) is PCLK2 = 170 MHz.
     *    With OVER8 = 0, BRR = f_CK / baud = 170000000 / 115200 = 1475.694...
     *    Rounded to nearest integer: 1476 (0x05C4)
     *    Actual baud = 170 MHz / 1476 = 115176.15 baud (~0.02% error).
     *    CRITICAL: BRR must be written before UE is set.

    /* 5. Enable Transmitter, Receiver, and the USART peripheral */
    USART1->CR1 |= USART_CR1_TE | USART_CR1_RE;
    USART1->CR1 |= USART_CR1_UE;

### Explanation

2. **Baud Rate Calculation:**
   - At reset, `RCC->CCIPR` routes PCLK2 (170 MHz) as the USART1 clock source.

     $$\text{BRR} = \text{round}\left(\frac{f_{\text{CK}}}{\text{Baud}}\right) = \text{round}\left(\frac{170\,000\,000}{115\,200}\right) = 1476 = \text{0x05C4}$$

3. **Register Sequencing:**
   - Setting `USART1->BRR` prior to enabling `USART_CR1_UE` fulfills the hardware requirement.
   - 8 data bits and no parity are selected by leaving $M[1:0] = 00$ and $PCE = 0$, while 1 stop bit is the default ($STOP[1:0] = 00$).

void I2C2_Init(void)
{
    /* 1. Enable peripheral clocks for GPIOA (AHB2) and I2C2 (APB1) */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_I2C2EN;

    /* 2. Configure PA8 (SDA) and PA9 (SCL):
     *    - Mode: Alternate Function (MODER = 10b)
     *    - Output type: Open-Drain (OT = 1)
     *    - Pull-up/Pull-down: Floating / No pull (PUPDR = 00b)
     *      CRITICAL: The external bus must be pulled up to the sensor's 1.8 V rail.
     *      Internal pull-ups must remain disabled to avoid pulling the bus to 3.3 V.
     *    - Alternate function: AF4 (I2C2) on both PA8 and PA9
     */
    GPIOA->MODER &= ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk);
    GPIOA->MODER |= (2U << GPIO_MODER_MODE8_Pos) | (2U << GPIO_MODER_MODE9_Pos);

    GPIOA->OTYPER |= GPIO_OTYPER_OT8 | GPIO_OTYPER_OT9;

    GPIOA->PUPDR &= ~(GPIO_PUPDR_PUPD8_Msk | GPIO_PUPDR_PUPD9_Msk);

    /* Assign AF4 to PA8 (AFRH bits 3:0) and PA9 (AFRH bits 7:4) */
    GPIOA->AFR[1] &= ~((0xFU << GPIO_AFRH_AFSEL8_Pos) | (0xFU << GPIO_AFRH_AFSEL9_Pos));
    GPIOA->AFR[1] |= (4U << GPIO_AFRH_AFSEL8_Pos) | (4U << GPIO_AFRH_AFSEL9_Pos);

    /* 3. Disable I2C2 before configuring TIMINGR */
    I2C2->CR1 &= ~I2C_CR1_PE;

    /* 4. Configure TIMINGR for Standard Mode (100 kHz):
     *    I2C2 clock source is PCLK1 = 170 MHz (t_I2CCLK ≈ 5.882 ns).
     *    - PRESC = 15 (0xF) => t_PRESC = (15 + 1) * 5.882 ns = 94.118 ns
     *    - SCLDEL = 4       => t_SCLDEL = (4 + 1) * t_PRESC = 470.6 ns (min 250 ns)
     *    - SDADEL = 2       => t_SDADEL = 2 * t_PRESC = 188.2 ns
     *    - SCLH = 49 (0x31) => t_HIGH = (49 + 1) * t_PRESC = 4.706 µs (min 4.0 µs)
     *    - SCLL = 55 (0x37) => t_LOW  = (55 + 1) * t_PRESC = 5.271 µs (min 4.7 µs)
     *    Total Period = (50 + 56) * 94.118 ns ≈ 9.976 µs (~100.24 kHz, < 0.25% error).
     */
    I2C2->TIMINGR = (15U << I2C_TIMINGR_PRESC_Pos)
                  | (4U  << I2C_TIMINGR_SCLDEL_Pos)
                  | (2U  << I2C_TIMINGR_SDADEL_Pos)
                  | (49U << I2C_TIMINGR_SCLH_Pos)
                  | (55U << I2C_TIMINGR_SCLL_Pos);

    /* 5. Enable I2C2 */
    I2C2->CR1 |= I2C_CR1_PE;
}
```

### Explanation
1. **1.8 V I/O Compatibility:** PA8 and PA9 are 5 V-tolerant (FT) pins. Configuring them as open-drain (`GPIO_OTYPER_OT`) without internal pull-ups (`PUPDR = 0`) allows external pull-up resistors to the sensor's $1.8\text{ V}$ supply to define the high logic level without driving $3.3\text{ V}$ into the sensor.
2. **Pin Assignment:** PA8 and PA9 are mapped to Alternate Function 4 (`AF4`), connecting them to `I2C2_SDA` and `I2C2_SCL` respectively.
3. **TIMINGR Calculation (100 kHz):** With $f_{\text{PCLK1}} = 170\text{ MHz}$, a prescaler of 16 gives a sub-clock period $t_{\text{PRESC}} \approx 94.12\text{ ns}$. Allocating 50 ticks to $t_{\text{HIGH}}$ ($4.71\ \mu\text{s}$) and 56 ticks to $t_{\text{LOW}}$ ($5.27\ \mu\text{s}$) satisfies $I^2C$ standard mode timing and yields $\approx 100.24\text{ kHz}$.
4. **Sequencing:** `I2C2->TIMINGR` is written while `PE = 0`, after which `I2C_CR1_PE` is asserted.

{

    /* AF2 configuration for PB4 in AFR[0] (AFRL) */
    GPIOB->AFR[0] &= ~(0xFU << GPIO_AFRL_AFSEL4_Pos);
    GPIOB->AFR[0] |= (2U << GPIO_AFRL_AFSEL4_Pos);

    /* 3. Configure TIM3 time base for 1 kHz:
     *    Timer clock is PCLK1 = 170 MHz (when APB1 prescaler is 1).
     *    Prescaler = 169  => Timer clock / (169 + 1) = 1 MHz (1 µs resolution)
     *    ARR = 999        => PWM frequency = 1 MHz / (999 + 1) = 1 kHz
     */
    TIM3->PSC = 170U - 1U;
    TIM3->ARR = 1000U - 1U;

    TIM3->CCR1 = 500U;

     *    OC1M[3:0] = 0110b (PWM Mode 1: active while CNT < CCR1)

    TIM3->CCMR1 |= (6U << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE;

     *    CC1P = 0 (active-high)
     *    CC1E = 1 (enable output to pin)

    TIM3->CCER |= TIM_CCER_CC1E;

    /* 7. Enable Auto-Reload Preload (ARPE) and trigger update event to load registers */
    TIM3->CR1 |= TIM_CR1_ARPE;
    TIM3->EGR = TIM_EGR_UG;

    /* 8. Start TIM3 counter */
    TIM3->CR1 |= TIM_CR1_CEN;
}

void UpdateClocksAndPeripherals_180MHz(void)
{

     * 1. Switch SYSCLK to 180 MHz
     * ========================================================================= */

    {
        /* Busy-wait until HSE is active SYSCLK */
    }

    /* Disable the PLL */
    RCC->CR &= ~RCC_CR_PLLON;
    while ((RCC->CR & RCC_CR_PLLRDY) != 0)
    {
        /* Busy-wait until PLL is fully stopped */
    }

    /* Increase Flash latency to 5 wait states for operation above 170 MHz */
    FLASH->ACR = (FLASH->ACR & ~FLASH_ACR_LATENCY_Msk) | FLASH_ACR_LATENCY_5WS;
    while ((FLASH->ACR & FLASH_ACR_LATENCY_Msk) != FLASH_ACR_LATENCY_5WS)
    {
        /* Busy-wait for Flash latency update */
    }

    /* Reconfigure PLL for 180 MHz:
     * Source: HSE (8 MHz)
     * M = 2  => 4 MHz input
     * N = 90 => VCO = 4 MHz * 90 = 360 MHz
     * R = 2  => PLLR = 360 MHz / 2 = 180 MHz
     */

                 | (1U << RCC_PLLCFGR_PLLM_Pos)      /* PLLM = 2 (reg value 1) */
                 | (90U << RCC_PLLCFGR_PLLN_Pos)     /* PLLN = 90 */
                 | (0U << RCC_PLLCFGR_PLLR_Pos)      /* PLLR = 2 (reg value 0) */
                 | RCC_PLLCFGR_PLLREN;

    /* Re-enable PLL */
    RCC->CR |= RCC_CR_PLLON;
    while ((RCC->CR & RCC_CR_PLLRDY) == 0)
    {
        /* Busy-wait for PLL lock */
    }

    /* Switch SYSCLK back to PLL */
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW_Msk) | RCC_CFGR_SW_PLL;
    while ((RCC->CFGR & RCC_CFGR_SWS_Msk) != RCC_CFGR_SWS_PLL)
    {
        /* Busy-wait until PLL is selected as SYSCLK */
    }

    /* =========================================================================
     * 2. Update USART1 Baud Rate (Target: 115200 baud, f_PCLK2 = 180 MHz)
     * ========================================================================= */

    USART1->CR1 &= ~USART_CR1_UE;

     * 3. Update I2C2 TIMINGR (Target: 100 kHz standard mode, f_PCLK1 = 180 MHz)

    /* Disable I2C2 before updating TIMINGR */
    I2C2->CR1 &= ~I2C_CR1_PE;

    /* PRESC = 15 => t_PRESC = 16 / 180 MHz = 88.889 ns

     * SDADEL = 2  => t_SDADEL = 2 * 88.889 ns = 177.8 ns
     * SCLH = 53   => t_HIGH   = (53 + 1) * 88.889 ns = 4.80 µs (min 4.0 µs)
     * SCLL = 58   => t_LOW    = (58 + 1) * 88.889 ns = 5.24 µs (min 4.7 µs)

                  | (4U  << I2C_TIMINGR_SCLDEL_Pos)
                  | (2U  << I2C_TIMINGR_SDADEL_Pos)
                  | (53U << I2C_TIMINGR_SCLH_Pos)
                  | (58U << I2C_TIMINGR_SCLL_Pos);

    I2C2->CR1 |= I2C_CR1_PE;

    /* =========================================================================
     * 4. Update TIM3 Prescaler (Target: 1 kHz PWM, 50% duty, f_CK_INT = 180 MHz)
     * ========================================================================= */

                 | (90U << RCC_PLLCFGR_PLLN_Pos)
                 | (0U  << RCC_PLLCFGR_PLLR_Pos)
                 | RCC_PLLCFGR_PLLREN;

    /* Enable PLL and wait for lock */
    RCC->CR |= RCC_CR_PLLON;
    while ((RCC->CR & RCC_CR_PLLRDY) == 0)
    {
        /* Busy-wait for PLL lock */
    }

    /* Keep AHB, APB1, APB2 prescalers at /1 */
    RCC->CFGR &= ~(RCC_CFGR_HPRE_Msk | RCC_CFGR_PPRE1_Msk | RCC_CFGR_PPRE2_Msk);

    /* Switch SYSCLK to PLL */
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW_Msk) | RCC_CFGR_SW_PLL;
    while ((RCC->CFGR & RCC_CFGR_SWS_Msk) != RCC_CFGR_SWS_PLL)
    {
        /* Busy-wait until PLL is system clock */
    }

    /* =========================================================================
     * 2. PERIPHERAL CLOCK GATING & GPIO CONFIGURATION
     * ========================================================================= */
    /* Enable GPIOA and GPIOB clocks */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN | RCC_AHB2ENR_GPIOBEN;

    /* Enable USART1 (APB2), I2C2 (APB1), and TIM3 (APB1) */
    RCC->APB2ENR  |= RCC_APB2ENR_USART1EN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_I2C2EN | RCC_APB1ENR1_TIM3EN;

    /*
     * Pin Assignment Note:
     * - PA8:  I2C2_SDA  (AF4, Open-Drain, No Pull)
     * - PA9:  I2C2_SCL  (AF4, Open-Drain, No Pull) [Conflicts with USART1_TX AF7]
     * - PA10: USART1_RX (AF7, Push-Pull, High Speed)
     * - PB4:  TIM3_CH1  (AF2, Push-Pull, High Speed)
     */

    /* Configure PA8, PA9, PA10 as Alternate Function */
    GPIOA->MODER &= ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk | GPIO_MODER_MODE10_Msk);
    GPIOA->MODER |= (2U << GPIO_MODER_MODE8_Pos)
                 |  (2U << GPIO_MODER_MODE9_Pos)
                 |  (2U << GPIO_MODER_MODE10_Pos);

    /* PA8 & PA9 Open-Drain for 1.8 V I2C (no internal pull-ups) */
    GPIOA->OTYPER |= GPIO_OTYPER_OT8 | GPIO_OTYPER_OT9;
    GPIOA->PUPDR  &= ~(GPIO_PUPDR_PUPD8_Msk | GPIO_PUPDR_PUPD9_Msk);

    /* PA8 = AF4 (I2C2_SDA), PA9 = AF4 (I2C2_SCL), PA10 = AF7 (USART1_RX) */
    GPIOA->AFR[1] &= ~((0xFU << GPIO_AFRH_AFSEL8_Pos)
                     | (0xFU << GPIO_AFRH_AFSEL9_Pos)
                     | (0xFU << GPIO_AFRH_AFSEL10_Pos));
    GPIOA->AFR[1] |= (4U << GPIO_AFRH_AFSEL8_Pos)
                  |  (4U << GPIO_AFRH_AFSEL9_Pos)
                  |  (7U << GPIO_AFRH_AFSEL10_Pos);

    /* Configure PB4 as AF2 (TIM3_CH1) */
    GPIOB->MODER &= ~GPIO_MODER_MODE4_Msk;
    GPIOB->MODER |= (2U << GPIO_MODER_MODE4_Pos);
    GPIOB->AFR[0] &= ~(0xFU << GPIO_AFRL_AFSEL4_Pos);
    GPIOB->AFR[0] |= (2U << GPIO_AFRL_AFSEL4_Pos);

    /* =========================================================================
     * 3. USART1 CONFIGURATION (115200 baud, 8N1, 16x oversampling)
     * ========================================================================= */
    USART1->CR1 = 0U;
    USART1->CR2 = 0U;
    USART1->CR3 = 0U;

    /* BRR = 180 MHz / 115200 ≈ 1563 (written before UE is set) */
    USART1->BRR = 1563U;
    USART1->CR1 |= USART_CR1_TE | USART_CR1_RE;
    USART1->CR1 |= USART_CR1_UE;

    /* =========================================================================
     * 4. I2C2 CONFIGURATION (Standard Mode 100 kHz)
     * ========================================================================= */
    I2C2->CR1 &= ~I2C_CR1_PE;

    /* TIMINGR calculated for PCLK1 = 180 MHz:
     * PRESC = 15 (divider 16 -> t_PRESC = 88.89 ns)
     * SCLDEL = 4, SDADEL = 2, SCLH = 53, SCLL = 58
     */
    I2C2->TIMINGR = (15U << I2C_TIMINGR_PRESC_Pos)
                  | (4U  << I2C_TIMINGR_SCLDEL_Pos)
                  | (2U  << I2C_TIMINGR_SDADEL_Pos)
                  | (53U << I2C_TIMINGR_SCLH_Pos)
                  | (58U << I2C_TIMINGR_SCLL_Pos);

    I2C2->CR1 |= I2C_CR1_PE;

    /* =========================================================================
     * 5. TIM3 CH1 PWM CONFIGURATION (PB4, 1 kHz, 50% duty cycle)
     * ========================================================================= */
    /* Prescaler: 180 MHz / 180 = 1 MHz count frequency */
    TIM3->PSC = 180U - 1U;
    TIM3->ARR = 1000U - 1U; /* 1 MHz / 1000 = 1 kHz */
    TIM3->CCR1 = 500U;      /* 50% duty cycle */

    /* PWM Mode 1, preload enabled on CH1 */
    TIM3->CCMR1 &= ~(TIM_CCMR1_OC1M_Msk | TIM_CCMR1_CC1S_Msk);
    TIM3->CCMR1 |= (6U << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE;

    /* Enable Channel 1 output, active-high */
    TIM3->CCER &= ~TIM_CCER_CC1P;
    TIM3->CCER |= TIM_CCER_CC1E;

    /* Enable Auto-reload preload and generate update event to latch shadow registers */
    TIM3->CR1 |= TIM_CR1_ARPE;
    TIM3->EGR = TIM_EGR_UG;

    /* Start TIM3 counter */
    TIM3->CR1 |= TIM_CR1_CEN;

    /* Main loop */
    while (1)
    {
    }
}
```

### Explanation
1. **Clock System:** Directly bootstraps the core from reset to $180\text{ MHz}$ by enabling Range 1 Boost Mode (`PWR_CR5_R1MODE = 0`), raising Flash read latency to 5 wait states (`FLASH_ACR_LATENCY_5WS`), locking the PLL to $8\text{ MHz} \times \frac{90}{2 \times 2} = 180\text{ MHz}$, and switching SYSCLK.
2. **Peripheral Adaptation:** All downstream peripheral timings are scaled for $f_{\text{PCLK1}} = f_{\text{PCLK2}} = 180\text{ MHz}$:
   - `USART1->BRR = 1563` keeps the baud rate at $115\,200$ ($-0.03\%$ error).
   - `I2C2->TIMINGR` prescales $180\text{ MHz}$ down to achieve standard mode $100\text{ kHz}$ ($99.56\text{ kHz}$, $0.44\%$ error). Open-drain mode without internal pull-ups prevents $3.3\text{ V}$ contention against the $1.8\text{ V}$ sensor bus.
   - `TIM3->PSC = 179` scales the timer count clock to $1\text{ MHz}$, yielding an exact $1\text{ kHz}$ output with $50\%$ duty cycle via `ARR = 999` and `CCR1 = 500`.
3. **Hardware Pin Allocation Notice:** As specified across the messages, pin `PA9` is multiplexed to both `USART1_TX` (AF7) and `I2C2_SCL` (AF4). In this unified initialization, `PA9` is set to `AF4` (`I2C2_SCL`) to allow the $I^2C$ pressure sensor clock line to function. To use debug TX simultaneously, `USART1_TX` must be mapped to an alternative pin (such as `PB6` or `PC4`).
