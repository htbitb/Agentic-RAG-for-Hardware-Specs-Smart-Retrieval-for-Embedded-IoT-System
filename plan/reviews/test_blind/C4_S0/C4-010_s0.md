/* 1. Enable GPIOB and TIM4 peripheral clocks */
RCC->AHB2ENR  |= RCC_AHB2ENR_GPIOBEN;
RCC->APB1ENR1 |= RCC_APB1ENR1_TIM4EN;

/* 2. Configure PB6 and PB7 for Alternate Function AF2 (TIM4_CH1, TIM4_CH2) */
// Set mode to Alternate Function (10b)
GPIOB->MODER &= ~(GPIO_MODER_MODE6_Msk | GPIO_MODER_MODE7_Msk);
GPIOB->MODER |=  (2U << GPIO_MODER_MODE6_Pos) | (2U << GPIO_MODER_MODE7_Pos);

// Set alternate function AF2 on PB6 and PB7
GPIOB->AFR[0] &= ~(GPIO_AFRL_AFSEL6_Msk | GPIO_AFRL_AFSEL7_Msk);
GPIOB->AFR[0] |=  (2U << GPIO_AFRL_AFSEL6_Pos) | (2U << GPIO_AFRL_AFSEL7_Pos);

// Set pin output speed to Very High
GPIOB->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED6_Pos) | (3U << GPIO_OSPEEDR_OSPEED7_Pos);

/* 3. Configure TIM4 time base */
// With APB1 prescaler = 1, timer clock frequency = PCLK1 = 170 MHz
// Target PWM frequency = 10 kHz -> Period = 170 MHz / 10 kHz = 17000 counts
TIM4->PSC = 0;              // Prescaler = 1 (counter clock = 170 MHz)
TIM4->ARR = 17000 - 1;      // Auto-reload value (16999)

/* 4. Configure Duty Cycles */
// Channel 1: 30% of 17000 = 5100
// Channel 2: 60% of 17000 = 10200
TIM4->CCR1 = 5100;
TIM4->CCR2 = 10200;

/* 5. Configure Channels 1 & 2 for PWM Mode 1 with preload enabled */
// OCxM = 0110b (PWM mode 1), OCxPE = 1 (Preload enable)
TIM4->CCMR1 = (6U << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE
            | (6U << TIM_CCMR1_OC2M_Pos) | TIM_CCMR1_OC2PE;

/* 6. Enable channel outputs (Active-high: CCxP = 0) */
TIM4->CCER = TIM_CCER_CC1E | TIM_CCER_CC2E;

/* 7. Enable Auto-reload preload (ARPE), trigger an update, and start timer */
TIM4->CR1 |= TIM_CR1_ARPE;
TIM4->EGR  = TIM_EGR_UG;    // Load prescaler and preload values
TIM4->CR1 |= TIM_CR1_CEN;   // Counter enable
```

### Explanation

* **Clock & Frequency Calculation:** When the APB1 prescaler is 1, the timer clock equals PCLK1 (170 MHz). To achieve a 10 kHz PWM frequency without prescaling, $\text{ARR} = \frac{170\text{ MHz}}{10\text{ kHz}} - 1 = 16999$.
* **Pin Routing:** PB6 and PB7 are configured to Alternate Function `AF2`, which maps them to `TIM4_CH1` and `TIM4_CH2` respectively.
* **PWM Mode 1:** In PWM mode 1 (`OCxM = 0110b`), the output is active when the counter is strictly less than `CCRx` and inactive otherwise. With `CCER` polarity bits set to `0`, outputs are active-high.
* **Preload & Counter Enable:** `OC1PE`, `OC2PE`, and `ARPE` buffer register updates to prevent glitches. `TIM_EGR_UG` transfers initial shadow registers before `TIM_CR1_CEN` enables the counter. TIM4 is a general-purpose timer, so break/dead-time configuration (`BDTR`) is not present.

/* 1. Enable USART1 and GPIOB peripheral clocks */
RCC->AHB2ENR |= RCC_AHB2ENR_GPIOBEN;
RCC->APB2ENR |= RCC_APB2ENR_USART1EN;

/* 2. Configure PB6 (TX) and PB7 (RX) for Alternate Function AF7 (USART1) */
// Set mode to Alternate Function (10b)
GPIOB->MODER &= ~(GPIO_MODER_MODE6_Msk | GPIO_MODER_MODE7_Msk);
GPIOB->MODER |=  (2U << GPIO_MODER_MODE6_Pos) | (2U << GPIO_MODER_MODE7_Pos);

// Set alternate function AF7 on PB6 and PB7
GPIOB->AFR[0] &= ~(GPIO_AFRL_AFSEL6_Msk | GPIO_AFRL_AFSEL7_Msk);
GPIOB->AFR[0] |=  (7U << GPIO_AFRL_AFSEL6_Pos) | (7U << GPIO_AFRL_AFSEL7_Pos);

// Set pin output speed to Very High for TX
GPIOB->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED6_Pos);

// Optional: Enable internal pull-up on RX (PB7) to prevent floating line when idle
GPIOB->PUPDR &= ~GPIO_PUPDR_PUPD7_Msk;
GPIOB->PUPDR |=  (1U << GPIO_PUPDR_PUPD7_Pos);

/* 3. Configure Baud Rate */
// USART1 clock source defaults to PCLK2 (RCC_CCIPR.USART1SEL = 00b) -> 170 MHz
// Oversampling by 16 (CR1.OVER8 = 0): BRR = f_CK / Baud = 170,000,000 / 115,200 ≈ 1475.694
// Round to nearest integer: BRR = 1476
USART1->BRR = 1476U;

/* 4. Configure frame format, enable Transmitter, Receiver, and USART */
// 8 data bits: M[1:0] = 00b (CR1 reset state)
// 1 stop bit:  STOP[1:0] = 00b (CR2 reset state)
// No parity:   PCE = 0 (CR1 reset state)
USART1->CR1 = USART_CR1_TE | USART_CR1_RE;  // Enable transmitter & receiver
USART1->CR1 |= USART_CR1_UE;                // Enable USART
```

### Explanation

* **Pin Mapping:** PB6 and PB7 are reconfigured from TIM4 to `AF7` to route to `USART1_TX` and `USART1_RX`.
* **Clock Source & Baud Rate:** In `RCC->CCIPR`, `USART1SEL` resets to `00b` (PCLK2 = 170 MHz). With default 16x oversampling (`OVER8 = 0` in `USART1->CR1`), the baud rate divisor is:
  $$\text{BRR} = \text{round}\left(\frac{170{,}000{,}000}{115{,}200}\right) = 1476$$
  The actual resulting baud rate is:
  $$\frac{170{,}000{,}000}{1476} \approx 115{,}176.15\text{ baud}$$
  This results in an error of approximately $-0.021\%$, well within the $\pm 2\%$ tolerance.
* **Sequencing:** `USART1->BRR` is written while `UE = 0`. Afterwards, `USART_CR1_TE` and `USART_CR1_RE` configure transmission and reception, followed by setting `USART_CR1_UE` to activate the peripheral.

### Assistant answer 2

```c
#include "stm32g4xx.h"

int main(void)
{
    /* ---------------------------------------------------------
     * 1. Enable Peripheral Clocks
     * --------------------------------------------------------- */
    RCC->AHB2ENR  |= RCC_AHB2ENR_GPIOBEN;   // Enable GPIOB clock
    RCC->APB1ENR1 |= RCC_APB1ENR1_TIM4EN;   // Enable TIM4 clock
    RCC->APB2ENR  |= RCC_APB2ENR_USART1EN;  // Enable USART1 clock

    /* ---------------------------------------------------------
     * 2. Configure TIM4 (10 kHz PWM, CH1 = 30%, CH2 = 60%)
     * --------------------------------------------------------- */
    // Time base: f_TIM = 170 MHz -> ARR = 170 MHz / 10 kHz - 1 = 16999
    TIM4->PSC = 0;
    TIM4->ARR = 17000 - 1;

    // Set compare values: CH1 = 30% (5100), CH2 = 60% (10200)
    TIM4->CCR1 = 5100;
    TIM4->CCR2 = 10200;

    // Configure Channels 1 & 2: PWM mode 1 (OCxM = 0110b), Preload enable (OCxPE = 1)
    TIM4->CCMR1 = (6U << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE
                | (6U << TIM_CCMR1_OC2M_Pos) | TIM_CCMR1_OC2PE;

    // Enable channel outputs (Active-high: CCxP = 0)
    TIM4->CCER = TIM_CCER_CC1E | TIM_CCER_CC2E;

    // Enable Auto-reload preload, generate update event, and start counter
    TIM4->CR1 |= TIM_CR1_ARPE;
    TIM4->EGR  = TIM_EGR_UG;
    TIM4->CR1 |= TIM_CR1_CEN;

    /* ---------------------------------------------------------
     * 3. Configure USART1 (115200 baud, 8N1, 16x oversampling)
     * --------------------------------------------------------- */
    // BRR = 170,000,000 / 115,200 ≈ 1476 (written before enabling USART)
    USART1->BRR = 1476U;

    // Enable Transmitter and Receiver
    USART1->CR1 = USART_CR1_TE | USART_CR1_RE;

    // Enable USART
    USART1->CR1 |= USART_CR1_UE;

    /* ---------------------------------------------------------
     * 4. GPIO Configuration (PB6 & PB7)
     * ---------------------------------------------------------
     * NOTE: PB6 and PB7 are shared between TIM4 (AF2) and USART1 (AF7).
     * Configuring PB6 and PB7 for AF7 connects them to USART1_TX and
     * USART1_RX. If AF2 is chosen instead, they output TIM4_CH1/CH2.
     * --------------------------------------------------------- */
    // Alternate function mode (10b)
    GPIOB->MODER &= ~(GPIO_MODER_MODE6_Msk | GPIO_MODER_MODE7_Msk);
    GPIOB->MODER |=  (2U << GPIO_MODER_MODE6_Pos) | (2U << GPIO_MODER_MODE7_Pos);

    // Speed: Very High for high-frequency switching
    GPIOB->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED6_Pos) | (3U << GPIO_OSPEEDR_OSPEED7_Pos);

    // Pull-up on PB7 (RX line idle high)
    GPIOB->PUPDR &= ~GPIO_PUPDR_PUPD7_Msk;
    GPIOB->PUPDR |=  (1U << GPIO_PUPDR_PUPD7_Pos);

    // Select Alternate Function:
    // AF7 (USART1): PB6 = USART1_TX, PB7 = USART1_RX
    // (To route TIM4 instead, load 2U into both fields)
    GPIOB->AFR[0] &= ~(GPIO_AFRL_AFSEL6_Msk | GPIO_AFRL_AFSEL7_Msk);
    GPIOB->AFR[0] |=  (7U << GPIO_AFRL_AFSEL6_Pos) | (7U << GPIO_AFRL_AFSEL7_Pos);

    while (1)
    {
    }
}
```

### Explanation

* **Peripheral Initialization:** 
  * `TIM4` operates from the 170 MHz PCLK1 clock with $\text{ARR} = 16999$ (10 kHz) and duty cycles at 5100 (30%) and 10200 (60%) in PWM Mode 1.
  * `USART1` operates from the 170 MHz PCLK2 clock with `BRR = 1476` (115200 baud, $-0.021\%$ error) and is configured for 8N1 before setting `UE`.
* **Hardware Resource Conflict on PB6/PB7:** 
  * In the STM32G474 pinout, PB6 and PB7 support both `TIM4_CH1`/`TIM4_CH2` via **AF2** and `USART1_TX`/`USART1_RX` via **AF7**. 
  * Because a physical pin multiplexer can only select one alternate function at a time, selecting `AF7` connects PB6 and PB7 to USART1. `TIM4` continues counting and generating output compare events internally, but its outputs will not be routed to external pins unless mapped to other pins (e.g., TIM4 on PB8/PB9 or USART1 on PA9/PA10).
