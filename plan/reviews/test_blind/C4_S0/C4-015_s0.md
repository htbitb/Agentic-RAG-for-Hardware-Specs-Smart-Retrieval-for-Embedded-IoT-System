#include "stm32g4xx.h"

void SystemClock_Config_170MHz(void)
{
    /* 1. Enable HSE oscillator and wait until it is ready */
    RCC->CR |= RCC_CR_HSEON;
    while ((RCC->CR & RCC_CR_HSERDY) == 0)
    {
        /* Wait for HSE ready */
    }

    /* 2. Enable Power Controller clock and enable Range 1 Boost Mode
     * Boost mode is required to reach frequencies above 150 MHz up to 170 MHz.
     * Reset value of VOS is Range 1 normal mode; clearing R1MODE engages Boost Mode.
     */
    RCC->APB1ENR1 |= RCC_APB1ENR1_PWREN;
    PWR->CR5 &= ~PWR_CR5_R1MODE;

    /* 3. Configure Flash memory latency
     * In Range 1 Boost mode at 170 MHz, 4 wait states (4 WS) are required.
     */
    FLASH->ACR = (FLASH->ACR & ~FLASH_ACR_LATENCY) | FLASH_ACR_LATENCY_4WS;
    while ((FLASH->ACR & FLASH_ACR_LATENCY) != FLASH_ACR_LATENCY_4WS)
    {
        /* Wait for latency setting to take effect */
    }

    /* 4. Configure PLL:
     * Source: HSE (8 MHz)
     * PLLM  = /2  -> 8 MHz / 2 = 4 MHz (VCO input, within allowed 2.66 - 16 MHz range)
     * PLLN  = x85 -> 4 MHz * 85 = 340 MHz (VCO output, within allowed 64 - 344 MHz range)
     * PLLR  = /2  -> 340 MHz / 2 = 170 MHz (PLLCLK)
     * Note: PLLM bitfield value is (divider - 1). PLLR bits 00b represent /2.
     */
    RCC->PLLCFGR = RCC_PLLCFGR_PLLSRC_HSE
                 | (1UL << RCC_PLLCFGR_PLLM_Pos)      /* PLLM = 2 (register value 1) */
                 | (85UL << RCC_PLLCFGR_PLLN_Pos)     /* PLLN = 85 */
                 | RCC_PLLCFGR_PLLREN;                /* Enable Main PLLCLK (PLL "R") output; PLLR = 00 (/2) */

    /* 5. Enable Main PLL and wait until it is locked */
    RCC->CR |= RCC_CR_PLLON;
    while ((RCC->CR & RCC_CR_PLLRDY) == 0)
    {
        /* Wait for PLL ready */
    }

    /* 6. Switch SYSCLK to PLL
     * AHB, APB1, and APB2 prescalers default to /1 at reset (HPRE=0, PPRE1=0, PPRE2=0).
     */
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW) | RCC_CFGR_SW_PLL;
    while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL)
    {
        /* Wait for system clock switch to PLL */
    }
}
```

### Explanation

1. **HSE & PLL Calculations**: With an 8.000 MHz HSE crystal, setting $M = 2$, $N = 85$, and $R = 2$ gives:
   $$\text{PLL}_{\text{IN}} = \frac{8\text{ MHz}}{2} = 4\text{ MHz}$$
   $$f_{\text{VCO}} = 4\text{ MHz} \times 85 = 340\text{ MHz}$$
   $$f_{\text{PLLCLK}} = \frac{340\text{ MHz}}{2} = 170.000\text{ MHz}$$
   This yields an exact 170 MHz clock (0.0% error) while remaining within all datasheet boundaries ($f_{\text{VCO}} \le 344\text{ MHz}$).
2. **Range 1 Boost Mode**: Operating at 170 MHz requires Range 1 Boost Mode. It is entered by clearing the `R1MODE` bit in `PWR->CR5`.
3. **Flash Wait States**: At 170 MHz under Range 1 Boost Mode ($V_{\text{CORE}}$ nominal), the Flash memory controller requires 4 wait states (`FLASH_ACR_LATENCY_4WS`) to read without errors.
4. **Prescalers**: At chip reset, `HPRE`, `PPRE1`, and `PPRE2` are all `0` (prescaler = 1), meaning:
   $$\text{SYSCLK} = \text{HCLK} = \text{PCLK1} = \text{PCLK2} = 170\text{ MHz}$$

#include "stm32g4xx.h"

void USART1_Init_115200(void)
{
    /* 1. Enable peripheral clocks for GPIOA and USART1 */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB2ENR |= RCC_APB2ENR_USART1EN;

    /* 2. Configure PA9 (TX) and PA10 (RX) as Alternate Function 7 (AF7)
     * - MODER: Set bits [19:18] for PA9 and [21:20] for PA10 to '10' (Alternate Function).
     * - AFRH: Set AFRH1 (PA9) and AFRH2 (PA10) to 7 (AF7 = USART1).
     * - OSPEEDR: Set PA9 to high/very-high speed (optional but recommended for TX).
     * - PUPDR: Reset state is floating, which is appropriate for USART.
     */
    GPIOA->MODER &= ~(GPIO_MODER_MODE9_Msk | GPIO_MODER_MODE10_Msk);
    GPIOA->MODER |= (2UL << GPIO_MODER_MODE9_Pos) | (2UL << GPIO_MODER_MODE10_Pos);

    GPIOA->AFR[1] &= ~((0x0FUL << GPIO_AFRH_AFSEL9_Pos) | (0x0FUL << GPIO_AFRH_AFSEL10_Pos));
    GPIOA->AFR[1] |= (7UL << GPIO_AFRH_AFSEL9_Pos) | (7UL << GPIO_AFRH_AFSEL10_Pos);

    GPIOA->OSPEEDR |= (3UL << GPIO_OSPEEDR_OSPEED9_Pos); /* Very high speed for PA9 */

    /* 3. Configure USART1:
     * By default at reset, RCC->CCIPR USART1SEL = 00 (PCLK2 selected, which is 170 MHz).
     * Oversampling is by 16 by default (OVER8 = 0 in CR1).
     * Data bits = 8, Parity = None by default (M1=0, M0=0, PCE=0 in CR1).
     * Stop bits = 1 by default (STOP[1:0] = 00 in CR2).
     */
    USART1->CR1 = 0;
    USART1->CR2 = 0;
    USART1->CR3 = 0;

    /* 4. Set the Baud Rate:
     * BRR = f_CK / Baud = 170000000 / 115200 = 1475.694 -> 1476 (rounded)
     * Resulting baud: 170000000 / 1476 = 115176 baud (0.02% error).
     * Written before USART is enabled.
     */
    USART1->BRR = 1476UL;

    /* 5. Enable Transmitter, Receiver, and the USART peripheral */
    USART1->CR1 |= USART_CR1_TE | USART_CR1_RE | USART_CR1_UE;
}
```

### Explanation

1. **Pin Multiplexing**: PA9 and PA10 map to AF7 for `USART1_TX` and `USART1_RX`, respectively. Pin alternate functions are selected via `GPIOA->AFR[1]` (AFRH register).
2. **Clock Source**: By default in `RCC->CCIPR`, `USART1SEL` is `00b`, which routes **PCLK2** (configured to 170 MHz in Message 1) as the USART1 clock input ($f_{\text{CK}} = 170\text{ MHz}$).
3. **Baud Rate Calculation**: With `OVER8 = 0` (oversampling by 16):
   $$\text{BRR} = \frac{f_{\text{CK}}}{\text{Baud}} = \frac{170\,000\,000}{115\,200} \approx 1475.694 \implies \text{USART1->BRR} = 1476$$
   The resulting baud rate is $\approx 115176.15\text{ baud}$ (only $0.02\%$ error, well within the allowable RS-232 $\pm 2\%$ budget).
4. **Ordering**: Per reference manual requirements, the baud rate register `BRR` is configured before setting `UE` (USART Enable) in `CR1`. Transmitter (`TE`) and Receiver (`RE`) are enabled along with `UE`.

#include "stm32g4xx.h"

void I2C2_Init_100kHz(void)
{
    /* 1. Enable clocks for GPIOA and I2C2 */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_I2C2EN;

    /* 2. Configure PA8 (SDA) and PA9 (SCL):
     * - Alternate Function 4 (AF4 = I2C2)
     * - Open-drain output (OTYPER)
     * - Floating / No internal pull-up (PUPDR = 00) because external pull-ups 
     *   to the sensor's 1.8 V rail must be used; internal pull-ups would pull to 3.3 V.
     */
    GPIOA->MODER &= ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk);
    GPIOA->MODER |= (2UL << GPIO_MODER_MODE8_Pos) | (2UL << GPIO_MODER_MODE9_Pos);

    GPIOA->OTYPER |= (GPIO_OTYPER_OT8 | GPIO_OTYPER_OT9);

    GPIOA->PUPDR &= ~(GPIO_PUPDR_PUPD8_Msk | GPIO_PUPDR_PUPD9_Msk);

    GPIOA->AFR[1] &= ~((0x0FUL << GPIO_AFRH_AFSEL8_Pos) | (0x0FUL << GPIO_AFRH_AFSEL9_Pos));
    GPIOA->AFR[1] |= (4UL << GPIO_AFRH_AFSEL8_Pos) | (4UL << GPIO_AFRH_AFSEL9_Pos);

    /* 3. Configure I2C2 Timing:
     * Clock source: PCLK1 = 170 MHz (reset default in RCC->CCIPR I2C2SEL = 00).
     *
     * Target: Standard Mode (100 kHz -> period = 10 us).
     * - PRESC  = 16: t_PRESC = (16 + 1) / 170 MHz = 100 ns.
     * - SCLDEL = 4:  Data setup time = (4 + 1) * 100 ns = 500 ns  (min is 250 ns).
     * - SDADEL = 2:  Data hold time  = 2 * 100 ns = 200 ns        (min is 0 ns).
     * - SCLH   = 47: High period     = (47 + 1) * 100 ns = 4.8 us (min is 4.0 us).
     * - SCLL   = 51: Low period      = (51 + 1) * 100 ns = 5.2 us (min is 4.7 us).
     * Total period = 4.8 us + 5.2 us = 10.0 us -> exactly 100.0 kHz (0.0% error).
     *
     * Ensure PE = 0 before writing TIMINGR (cleared at reset).
     */
    I2C2->CR1 &= ~I2C_CR1_PE;

    I2C2->TIMINGR = (16UL << I2C_TIMINGR_PRESC_Pos)
                  | (4UL  << I2C_TIMINGR_SCLDEL_Pos)
                  | (2UL  << I2C_TIMINGR_SDADEL_Pos)
                  | (47UL << I2C_TIMINGR_SCLH_Pos)
                  | (51UL << I2C_TIMINGR_SCLL_Pos);

    /* 4. Enable I2C2 peripheral */
    I2C2->CR1 |= I2C_CR1_PE;
}
```

### Explanation

1. **Pin Multiplexing & 1.8 V Interfacing**: 
   - PA8 (`I2C2_SDA`) and PA9 (`I2C2_SCL`) are mapped to Alternate Function 4 (AF4).
   - PA8 and PA9 are Five-Volt Tolerant (FT) pins. When set to **open-drain** mode with internal pull-ups **disabled**, they safely interface with an external 1.8 V I/O rail using external pull-up resistors to 1.8 V.
2. **Clock Source**: By default in `RCC->CCIPR`, `I2C2SEL = 00b`, connecting I2C2 to **PCLK1** ($f_{\text{I2CCLK}} = 170\text{ MHz}$).
3. **Timing Calculation**:
   - Prescaler: $\text{PRESC} = 16 \implies t_{\text{PRESC}} = \frac{16 + 1}{170\text{ MHz}} = 100\text{ ns}$.
   - SCL clock period: 
     $$t_{\text{SCL}} = (t_{\text{SCLL}} + 1 + t_{\text{SCLH}} + 1) \times t_{\text{PRESC}} = (52 + 48) \times 100\text{ ns} = 10\,\mu\text{s} \implies 100\text{ kHz}$$
   - Both high ($4.8\,\mu\text{s}$) and low ($5.2\,\mu\text{s}$) durations satisfy the $I^2C$ Standard Mode minimum requirements ($t_{\text{HIGH}} \ge 4.0\,\mu\text{s}$, $t_{\text{LOW}} \ge 4.7\,\mu\text{s}$).

#include "stm32g4xx.h"

void TIM3_CH1_PWM_Init_1kHz(void)
{
    /* 1. Enable peripheral clocks for GPIOB and TIM3 */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOBEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_TIM3EN;

    /* 2. Configure PB4 as Alternate Function 2 (AF2 = TIM3_CH1)
     * Note: PB4 defaults to NJTRST after reset; writing MODER switches it to AF.
     */
    GPIOB->MODER &= ~GPIO_MODER_MODE4_Msk;
    GPIOB->MODER |= (2UL << GPIO_MODER_MODE4_Pos);

    GPIOB->AFR[0] &= ~(0x0FUL << GPIO_AFRL_AFSEL4_Pos);
    GPIOB->AFR[0] |= (2UL << GPIO_AFRL_AFSEL4_Pos);

    /* 3. Configure Time Base:
     * APB1 timer clock = PCLK1 = 170 MHz (APB1 prescaler is 1).
     * TIM3 is a 16-bit timer.
     * Prescaler = 16 -> f_counter = 170 MHz / (16 + 1) = 10 MHz.
     * Period (ARR) = 9999 -> f_PWM = 10 MHz / (9999 + 1) = 1000 Hz (1 kHz).
     * Duty cycle 50%: CCR1 = (ARR + 1) * 50% = 5000.
     */
    TIM3->PSC = 16UL;
    TIM3->ARR = 9999UL;
    TIM3->CCR1 = 5000UL;

    /* 4. Configure Channel 1 for PWM Mode 1:
     * - CC1S = 00 (Channel 1 configured as output)
     * - OC1M = 0110b (PWM Mode 1: active while CNT < CCR1, inactive otherwise)
     * - OC1PE = 1 (Output compare 1 preload enable)
     */
    TIM3->CCMR1 &= ~(TIM_CCMR1_CC1S | TIM_CCMR1_OC1M);
    TIM3->CCMR1 |= (6UL << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE;

    /* 5. Configure Output Polarity and Enable Channel 1:
     * - CC1P = 0 (Active high)
     * - CC1E = 1 (OC1 output enabled to the pin)
     */
    TIM3->CCER &= ~TIM_CCER_CC1P;
    TIM3->CCER |= TIM_CCER_CC1E;

    /* 6. Enable auto-reload preload and generate update event to load shadow registers */
    TIM3->CR1 |= TIM_CR1_ARPE;
    TIM3->EGR |= TIM_EGR_UG;

    /* 7. Start the counter */
    TIM3->CR1 |= TIM_CR1_CEN;
}
```

### Explanation

1. **Pin Mapping**: PB4 routes to `TIM3_CH1` via Alternate Function 2 (`AF2`). Because PB4 is multiplexed with JTAG `NJTRST` at reset, setting `MODER` to alternate function reclaims the pin for timer output.
2. **Timer Clock**: Since the APB1 prescaler is 1, the timer clock input to TIM3 equals `PCLK1` (170 MHz).
3. **Frequency and Duty Cycle Calculations**:
   - Prescaler: $\text{PSC} = 16 \implies f_{\text{CNT}} = \frac{170\text{ MHz}}{16 + 1} = 10\text{ MHz}$.
   - Auto-reload: $\text{ARR} = 9999 \implies f_{\text{PWM}} = \frac{10\text{ MHz}}{9999 + 1} = 1.000\text{ kHz}$ (0.0% error).
   - Duty cycle: $\text{CCR1} = 5000 \implies \frac{5000}{10000} = 50.0\%$.
4. **PWM Mode 1**: Configured with `OC1M = 0110b` and `CC1P = 0` (active high), meaning the output stays high while `TIM3_CNT < TIM3_CCR1` and goes low when `TIM3_CNT >= TIM3_CCR1`.

### Assistant answer 4

> **Note on device specifications:** The maximum rated core frequency of the STM32G474 is **170 MHz** in Range 1 Boost mode [DS12288 p.1], and the maximum rated PLL VCO frequency is **344 MHz** [DS12288 p.124]. ### C Implementation

```c
#include "stm32g4xx.h"

void Clock_And_Peripherals_Update_180MHz(void)
{
    /* 1. Ensure HSI16 is ON to serve as temporary clock */
    RCC->CR |= RCC_CR_HSION;
    while ((RCC->CR & RCC_CR_HSIRDY) == 0)
    {
        /* Wait for HSI ready */
    }

    /* 2. Switch SYSCLK to HSI16 */
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW) | RCC_CFGR_SW_HSI;
    while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_HSI)
    {
        /* Wait for switch to HSI */
    }

    /* 3. Disable the PLL before reprogramming */
    RCC->CR &= ~RCC_CR_PLLON;
    while ((RCC->CR & RCC_CR_PLLRDY) != 0)
    {
        /* Wait for PLL to stop */
    }

    /* 4. Configure PLL for 180 MHz:
     * Source: HSE (8 MHz)
     * PLLM = 2  -> 8 MHz / 2 = 4 MHz
     * PLLN = 90 -> 4 MHz * 90 = 360 MHz VCO
     * PLLR = /2 -> 360 MHz / 2 = 180 MHz (PLLCLK)
     */
    RCC->PLLCFGR = RCC_PLLCFGR_PLLSRC_HSE
                 | (1UL << RCC_PLLCFGR_PLLM_Pos)      /* PLLM = 2 */
                 | (90UL << RCC_PLLCFGR_PLLN_Pos)     /* PLLN = 90 */
                 | RCC_PLLCFGR_PLLREN;                /* PLLR = /2 */

    /* 5. Enable PLL and wait for lock */
    RCC->CR |= RCC_CR_PLLON;
    while ((RCC->CR & RCC_CR_PLLRDY) == 0)
    {
        /* Wait for PLL ready */
    }

    /* 6. Switch SYSCLK to PLL (180 MHz) */
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW) | RCC_CFGR_SW_PLL;
    while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL)
    {
        /* Wait for system clock switch */
    }

    /* ---------------- Update Peripherals for 180 MHz ---------------- */

    /* 7. Update USART1 for 115200 baud (PCLK2 = 180 MHz):
     * BRR = 180,000,000 / 115,200 = 1562.5 -> 1563 (0.03% error)
     */
    USART1->CR1 &= ~USART_CR1_UE;
    USART1->BRR = 1563UL;
    USART1->CR1 |= USART_CR1_UE;

    /* 8. Update I2C2 for 100 kHz SCL (PCLK1 = 180 MHz):
     * PRESC = 17 -> t_PRESC = (17 + 1) / 180 MHz = 100 ns.
     * SCLH = 47 (4.8 us), SCLL = 51 (5.2 us) -> period = 10.0 us (100.0 kHz).
     */
    I2C2->CR1 &= ~I2C_CR1_PE;
    I2C2->TIMINGR = (17UL << I2C_TIMINGR_PRESC_Pos)
                  | (4UL  << I2C_TIMINGR_SCLDEL_Pos)
                  | (2UL  << I2C_TIMINGR_SDADEL_Pos)
                  | (47UL << I2C_TIMINGR_SCLH_Pos)
                  | (51UL << I2C_TIMINGR_SCLL_Pos);
    I2C2->CR1 |= I2C_CR1_PE;

    /* 9. Update TIM3 for 1 kHz PWM (PCLK1 = 180 MHz):
     * PSC = 17 -> f_CNT = 180 MHz / (17 + 1) = 10 MHz.
     * ARR remains 9999 (1 kHz) and CCR1 remains 5000 (50% duty cycle).
     */
    TIM3->PSC = 17UL;
    TIM3->EGR |= TIM_EGR_UG; /* Reload prescaler shadow register immediately */
}
```

### Explanation

1. **PLL Reconfiguration Sequence**: `RCC->PLLCFGR` can only be written when the PLL is disabled. SYSCLK is temporarily switched back to HSI16 before clearing `PLLON`, reprogramming $N=90$, relocking the PLL, and switching back.
2. **USART1**: With $f_{\text{CK}} = 180\text{ MHz}$, $\text{BRR} = 180\,000\,000 / 115\,200 \approx 1563$, maintaining $115200\text{ baud}$ ($0.03\%$ error).
3. **I2C2**: Incrementing `PRESC` from 16 to 17 maintains an identical clock slice $t_{\text{PRESC}} = \frac{18}{180\text{ MHz}} = 100\text{ ns}$, keeping the exact 100.0 kHz SCL frequency and timing profiles.
4. **TIM3**: Incrementing `PSC` from 16 to 17 preserves the counter clock at $10\text{ MHz}$, keeping the output frequency at $1\text{ kHz}$ and duty cycle at $50\%$.

    GPIOA->AFR[1] |= (7UL << GPIO_AFRH_AFSEL10_Pos);

    /* Configure USART1: PCLK2 = 180 MHz, OVER8 = 0
     * BRR = 180,000,000 / 115,200 = 1562.5 -> 1563
     */
    USART1->CR1 = 0;
    USART1->CR2 = 0;
    USART1->CR3 = 0;
    USART1->BRR = 1563UL;
    USART1->CR1 |= USART_CR1_TE | USART_CR1_RE | USART_CR1_UE;

    /* =========================================================================
     * 4. I2C2 CONFIGURATION (Standard Mode 100 kHz, PA8=SDA, PA9=SCL, 1.8V bus)
     * ========================================================================= */
    /* Configure PA8 (SDA) and PA9 (SCL): AF4, Open-Drain, No Pull */
    GPIOA->MODER &= ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk);
    GPIOA->MODER |= (2UL << GPIO_MODER_MODE8_Pos) | (2UL << GPIO_MODER_MODE9_Pos);

    GPIOA->OTYPER |= (GPIO_OTYPER_OT8 | GPIO_OTYPER_OT9);
    GPIOA->PUPDR  &= ~(GPIO_PUPDR_PUPD8_Msk | GPIO_PUPDR_PUPD9_Msk);

    GPIOA->AFR[1] &= ~((0x0FUL << GPIO_AFRH_AFSEL8_Pos) | (0x0FUL << GPIO_AFRH_AFSEL9_Pos));
    GPIOA->AFR[1] |= (4UL << GPIO_AFRH_AFSEL8_Pos) | (4UL << GPIO_AFRH_AFSEL9_Pos);

    /* Configure I2C2 Timing: PCLK1 = 180 MHz
     * PRESC = 17 -> t_PRESC = 100 ns
     * SCLDEL = 4 (500 ns), SDADEL = 2 (200 ns)
     * SCLH = 47 (4.8 us), SCLL = 51 (5.2 us) -> period = 10.0 us (100.0 kHz)
     */
    I2C2->CR1 &= ~I2C_CR1_PE;
    I2C2->TIMINGR = (17UL << I2C_TIMINGR_PRESC_Pos)
                  | (4UL  << I2C_TIMINGR_SCLDEL_Pos)
                  | (2UL  << I2C_TIMINGR_SDADEL_Pos)
                  | (47UL << I2C_TIMINGR_SCLH_Pos)
                  | (51UL << I2C_TIMINGR_SCLL_Pos);
    I2C2->CR1 |= I2C_CR1_PE;

    /* =========================================================================
     * 5. TIM3 PWM CONFIGURATION (1 kHz, 50% duty cycle on PB4 / TIM3_CH1)
     * ========================================================================= */
    /* Configure PB4 as AF2 (TIM3_CH1) */
    GPIOB->MODER &= ~GPIO_MODER_MODE4_Msk;
    GPIOB->MODER |= (2UL << GPIO_MODER_MODE4_Pos);
    GPIOB->AFR[0] &= ~(0x0FUL << GPIO_AFRL_AFSEL4_Pos);
    GPIOB->AFR[0] |= (2UL << GPIO_AFRL_AFSEL4_Pos);

    /* APB1 Timer Clock = 180 MHz
     * PSC = 17 -> f_CNT = 10 MHz
     * ARR = 9999 -> f_PWM = 10 MHz / 10000 = 1000 Hz
     * CCR1 = 5000 -> 50% duty cycle
     */
    TIM3->PSC  = 17UL;
    TIM3->ARR  = 9999UL;
    TIM3->CCR1 = 5000UL;

    /* PWM Mode 1, preload enabled, active-high, output enabled */
    TIM3->CCMR1 &= ~(TIM_CCMR1_CC1S | TIM_CCMR1_OC1M);
    TIM3->CCMR1 |= (6UL << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE;
    TIM3->CCER  &= ~TIM_CCER_CC1P;
    TIM3->CCER  |= TIM_CCER_CC1E;
    TIM3->CR1   |= TIM_CR1_ARPE;

    /* Load preload registers and start TIM3 */
    TIM3->EGR |= TIM_EGR_UG;
    TIM3->CR1 |= TIM_CR1_CEN;

    /* Main loop */
    while (1)
    {
    }
}

### Explanation

1. **Direct 180 MHz Bring-Up**: Since the MCU executes `main()` after reset running on HSI16, the clock tree transitions directly to 180 MHz by enabling HSE, selecting Range 1 Boost mode, setting 4 Flash wait states, and locking the PLL at $8\text{ MHz} \times \frac{90}{2 \times 2} = 180\text{ MHz}$.
2. **Peripheral Adaptation (180 MHz)**:
   - **USART1**: $\text{BRR} = \frac{180\,000\,000}{115\,200} \approx 1563$ ($115\,163\text{ baud}$, $0.03\%$ error).
   - **I2C2**: Setting $\text{PRESC} = 17$ scales the $180\text{ MHz}$ PCLK1 clock to a $100\text{ ns}$ base tick, matching the previous $100.0\text{ kHz}$ SCL waveform.
   - **TIM3**: Setting $\text{PSC} = 17$ scales $180\text{ MHz}$ to a $10\text{ MHz}$ counter clock, preserving the $1.0\text{ kHz}$ PWM frequency with $\text{ARR} = 9999$.
3. **Pin Hardware Note (PA9)**: Per your configuration, **PA9** was specified for both `USART1_TX` (AF7) and `I2C2_SCL` (AF4). In this program, PA9 is assigned to `I2C2_SCL` (AF4) in open-drain mode to satisfy the 1.8 V I2C bus requirement. If USART1 bidirectional communication is needed, USART1 TX must be remapped to another pin (e.g., `PB6` AF7 or `PC4` AF7).
