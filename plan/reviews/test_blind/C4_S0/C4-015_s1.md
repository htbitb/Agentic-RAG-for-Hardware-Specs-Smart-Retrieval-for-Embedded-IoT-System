#include "stm32g4xx.h"

void I2C2_Init(void)
{
    /* 1. Enable peripheral clocks for GPIOA and I2C2 */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_I2C2EN;

    /* 2. Configure PA8 (SDA) and PA9 (SCL):
     *    - Alternate Function mode (AF4)
     *    - Open-drain output (mandatory for I2C and ensures lines are pulled
     *      up to the sensor's 1.8 V rail via external pull-ups, never to 3.3 V)
     *    - No internal pull-up / pull-down (floating)
     *    - High output speed
     */
    GPIOA->MODER = (GPIOA->MODER & ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk))
                 | (2U << GPIO_MODER_MODE8_Pos)
                 | (2U << GPIO_MODER_MODE9_Pos);

    GPIOA->OTYPER |= GPIO_OTYPER_OT8 | GPIO_OTYPER_OT9;

    GPIOA->PUPDR &= ~(GPIO_PUPDR_PUPD8_Msk | GPIO_PUPDR_PUPD9_Msk);

    GPIOA->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED8_Pos)
                   |  (3U << GPIO_OSPEEDR_OSPEED9_Pos);

    /* AF4 on PA8 (I2C2_SDA) and PA9 (I2C2_SCL) */
    GPIOA->AFR[1] = (GPIOA->AFR[1] & ~(GPIO_AFRH_AFSEL8_Msk | GPIO_AFRH_AFSEL9_Msk))
                  | (4U << GPIO_AFRH_AFSEL8_Pos)
                  | (4U << GPIO_AFRH_AFSEL9_Pos);

    /* 3. Make sure I2C2 is disabled before programming TIMINGR */
    I2C2->CR1 &= ~I2C_CR1_PE;

    /* 4. Configure I2C2 timing for 100 kHz Standard Mode from PCLK1 = 170 MHz:
     *    t_I2CCLK = 1 / 170 MHz = 5.882 ns
     *    PRESC  = 15  -> t_PRESC = (15 + 1) * 5.882 ns = 94.118 ns
     *    SCLL   = 55  -> t_LOW   = (55 + 1) * 94.118 ns = 5.27 us  (min 4.7 us)
     *    SCLH   = 49  -> t_HIGH  = (49 + 1) * 94.118 ns = 4.71 us  (min 4.0 us)
     *    SDADEL = 2   -> t_HD;DAT = 2 * 94.118 ns       = 188.2 ns (min 0 ns)
     *    SCLDEL = 4   -> t_SU;DAT = (4 + 1) * 94.118 ns = 470.6 ns (min 250 ns)
     *    Total Period = (56 + 50) * 94.118 ns = 9.977 us -> ~100.2 kHz
     */
    I2C2->TIMINGR = (15U << I2C_TIMINGR_PRESC_Pos)
                  | (4U  << I2C_TIMINGR_SCLDEL_Pos)
                  | (2U  << I2C_TIMINGR_SDADEL_Pos)
                  | (49U << I2C_TIMINGR_SCLH_Pos)
                  | (55U << I2C_TIMINGR_SCLL_Pos);

    /* 5. Enable I2C2 peripheral */
    I2C2->CR1 |= I2C_CR1_PE;
}
```

### Explanation

1. **1.8 V Supply Consideration**:
   - STM32 5V-tolerant (FT) pins have no high-side clamping diodes to $V_{DD}$ when operating in open-drain mode.
   - Configuring PA8 and PA9 as **open-drain** (`OTYPER` bits set) with **no internal pull-ups** (`PUPDR = 00b`) ensures the lines are only driven low to $0\text{ V}$ and rely exclusively on external pull-up resistors to the sensor's $1.8\text{ V}$ bus rail.
2. **Alternate Function**:
   - Both PA8 (I2C2_SDA) and PA9 (I2C2_SCL) map to Alternate Function 4 (`AF4`).
3. **Timing Calculation ($f_{\text{PCLK1}} = 170\text{ MHz}$)**:
   - Prescaler divider is set to $\text{PRESC} + 1 = 16$, giving a prescaled clock period $t_{\text{PRESC}} \approx 94.118\text{ ns}$.
   - Count values $\text{SCLL} = 55$ and $\text{SCLH} = 49$ yield a frequency of $100.23\text{ kHz}$, satisfying the $100\text{ kHz}$ Standard Mode timing constraints ($t_{\text{LOW}} \ge 4.7\ \mu\text{s}$, $t_{\text{HIGH}} \ge 4.0\ \mu\text{s}$, $t_{\text{SU;DAT}} \ge 250\text{ ns}$).
   - `TIMINGR` is written while `PE = 0`, as required by the I2C hardware architecture [RM0440 p.1023].

void StatusLED_PWM_Init(void)
{

    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOBEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_TIM3EN;

    /* 2. Configure PB4 for Alternate Function 2 (TIM3_CH1) */
    /* Set Alternate Function mode (10b) */
    GPIOB->MODER = (GPIOB->MODER & ~GPIO_MODER_MODE4_Msk)
                 | (2U << GPIO_MODER_MODE4_Pos);

    /* Output type: Push-Pull (reset default = 0) */
    GPIOB->OTYPER &= ~GPIO_OTYPER_OT4;

    /* Output speed: High */
    GPIOB->OSPEEDR |= (2U << GPIO_OSPEEDR_OSPEED4_Pos);

    /* Set AF2 in AFR[0] (AFRL) for pin 4 */
    GPIOB->AFR[0] = (GPIOB->AFR[0] & ~GPIO_AFRL_AFSEL4_Msk)
                  | (2U << GPIO_AFRL_AFSEL4_Pos);

    /* 3. Configure TIM3 time base for 1 kHz frequency:
     *    TIM3 clock is 170 MHz (PCLK1 prescaler = 1).
     *    Prescaler = 169 -> timer counter clock = 170 MHz / (169 + 1) = 1 MHz
     *    ARR = 999       -> PWM period = 1 MHz / (999 + 1) = 1000 Hz (1 kHz)
     */
    TIM3->PSC = 169U;
    TIM3->ARR = 999U;

    /* 4. Configure 50 % duty cycle:
     *    CCR1 = 500 (active for 500 out of 1000 ticks)
     */
    TIM3->CCR1 = 500U;

    /* 5. Configure Channel 1:
     *    - PWM Mode 1: OC1M = 0110b (active when CNT < CCR1)
     *    - Output preload enable: OC1PE = 1
     */
    TIM3->CCMR1 = (6U << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE;

    /* 6. Enable Channel 1 output, active-high (CC1P = 0 is active-high) */
    TIM3->CCER |= TIM_CCER_CC1E;

    /* 7. Enable Auto-reload preload (ARPE) and trigger update event to load shadow registers */
    TIM3->CR1 |= TIM_CR1_ARPE;
    TIM3->EGR |= TIM_EGR_UG;

    /* 8. Start TIM3 counter */
    TIM3->CR1 |= TIM_CR1_CEN;
}
```

### Explanation

1. **GPIO Alternate Function**:
   - Pin PB4 maps to Alternate Function 2 (`AF2`) for `TIM3_CH1` [RM0440 p.307].
2. **Frequency and Duty Cycle**:
   - With an APB1 prescaler of 1, the timer counter clock is $f_{\text{TIMCLK}} = f_{\text{PCLK1}} = 170\text{ MHz}$.
   - Setting $\text{PSC} = 169$ scales the clock to $1\text{ MHz}$ ($1\ \mu\text{s}$ per tick).
   - Setting $\text{ARR} = 999$ creates a 1000-tick period ($1\text{ ms} = 1\text{ kHz}$).
   - $\text{CCR1} = 500$ produces an active-high output for 500 ticks and low for 500 ticks ($50\%$ duty cycle).
3. **Timer Modes**:
   - **PWM Mode 1** (`OC1M = 0110b`): The output is high when $\text{CNT} < \text{CCR1}$ and low otherwise.
   - Preload registers (`OC1PE` and `ARPE`) prevent glitches during updates.
   - Initial update generation (`EGR_UG`) forces immediate loading of prescaler and preload registers.

    }

    /* 2. Disable the PLL */
    RCC->CR &= ~RCC_CR_PLLON;
    while ((RCC->CR & RCC_CR_PLLRDY) != 0)
    {
        /* Wait for PLL to stop */
    }

    /* 3. Increase Flash latency to 5 wait states for 180 MHz operation */
    FLASH->ACR = (FLASH->ACR & ~FLASH_ACR_LATENCY) | FLASH_ACR_LATENCY_5WS;

    {
        /* Wait for Flash latency update */
    }

    /* 4. Configure PLL for 180 MHz:
     *    Source: HSE = 8 MHz
     *    PLLM = 2  (8 MHz / 2 = 4 MHz input to VCO)
     *    PLLN = 90 (4 MHz * 90 = 360 MHz VCO)

> **Note on Device Limits**: According to the STM32G474 datasheet, the rated maximum frequency for $f_{\text{HCLK}}$ is **170 MHz** in Range 1 Boost mode, and the maximum recommended VCO output frequency is **344 MHz** [DS12288 p.76, p.122].     GPIOB->OSPEEDR|= (2U << GPIO_OSPEEDR_OSPEED4_Pos);       /* High Speed */
    GPIOB->AFR[0]  = (GPIOB->AFR[0] & ~GPIO_AFRL_AFSEL4_Msk)
                   | (2U << GPIO_AFRL_AFSEL4_Pos);           /* AF2 */

    /* --- Port A: PA8 (SDA), PA9 (SCL), PA10 (RX) --- */
    GPIOA->MODER   = (GPIOA->MODER & ~(GPIO_MODER_MODE8_Msk | GPIO_MODER_MODE9_Msk | GPIO_MODER_MODE10_Msk))
                   | (2U << GPIO_MODER_MODE8_Pos)
                   | (2U << GPIO_MODER_MODE9_Pos)
                   | (2U << GPIO_MODER_MODE10_Pos);          /* AF mode for PA8, PA9, PA10 */

    /* PA8 & PA9 Open-Drain for 1.8 V I2C (no internal pull-up/down) */
    GPIOA->OTYPER |= GPIO_OTYPER_OT8 | GPIO_OTYPER_OT9;
    GPIOA->PUPDR  &= ~(GPIO_PUPDR_PUPD8_Msk | GPIO_PUPDR_PUPD9_Msk | GPIO_PUPDR_PUPD10_Msk);

    GPIOA->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED8_Pos)
                   |  (3U << GPIO_OSPEEDR_OSPEED9_Pos);

    /* Alternate functions: PA8 -> AF4 (I2C2), PA9 -> AF4 (I2C2), PA10 -> AF7 (USART1) */
    GPIOA->AFR[1]  = (GPIOA->AFR[1] & ~(GPIO_AFRH_AFSEL8_Msk | GPIO_AFRH_AFSEL9_Msk | GPIO_AFRH_AFSEL10_Msk))
                   | (4U << GPIO_AFRH_AFSEL8_Pos)
                   | (4U << GPIO_AFRH_AFSEL9_Pos)
                   | (7U << GPIO_AFRH_AFSEL10_Pos);

    /* =========================================================================
     * 4. USART1 CONFIGURATION (115200 baud @ 180 MHz)
     * ========================================================================= */
    USART1->CR1 &= ~USART_CR1_UE;
    USART1->BRR = 1563U;                                     /* 180 MHz / 115200 ≈ 1562.5 -> 1563 */
    USART1->CR1 = USART_CR1_TE | USART_CR1_RE;               /* 8 data bits, no parity, TE, RE */
    USART1->CR1 |= USART_CR1_UE;

    /* =========================================================================
     * 5. I2C2 CONFIGURATION (Standard Mode 100 kHz @ 180 MHz)
     * ========================================================================= */
    I2C2->CR1 &= ~I2C_CR1_PE;
    I2C2->TIMINGR = (17U << I2C_TIMINGR_PRESC_Pos)           /* t_PRESC = 100 ns */
                  | (4U  << I2C_TIMINGR_SCLDEL_Pos)          /* t_SU;DAT = 500 ns */
                  | (2U  << I2C_TIMINGR_SDADEL_Pos)          /* t_HD;DAT = 200 ns */
                  | (46U << I2C_TIMINGR_SCLH_Pos)            /* t_HIGH = 4.7 us */
                  | (52U << I2C_TIMINGR_SCLL_Pos);           /* t_LOW  = 5.3 us -> Total 10.0 us (100 kHz) */
    I2C2->CR1 |= I2C_CR1_PE;

    /* =========================================================================
     * 6. TIM3 PWM CONFIGURATION (1 kHz, 50 % duty cycle @ 180 MHz)
     * ========================================================================= */
    TIM3->PSC   = 179U;                                      /* 180 MHz / (179 + 1) = 1 MHz counter clock */
    TIM3->ARR   = 999U;                                      /* 1 MHz / (999 + 1) = 1 kHz PWM period */
    TIM3->CCR1  = 500U;                                      /* 500 / 1000 = 50 % duty cycle */
    TIM3->CCMR1 = (6U << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE; /* PWM Mode 1, Preload enabled */
    TIM3->CCER |= TIM_CCER_CC1E;                             /* Enable Channel 1 output */
    TIM3->CR1  |= TIM_CR1_ARPE;                              /* Enable Auto-reload preload */
    TIM3->EGR  |= TIM_EGR_UG;                                /* Generate update to load registers */
    TIM3->CR1  |= TIM_CR1_CEN;                               /* Start TIM3 counter */

    /* =========================================================================
     * 7. MAIN LOOP
     * ========================================================================= */
    while (1)
    {
    }
}

### Explanation

1. **Clock Configuration (180 MHz)**:
   - Configures Range 1 Boost Mode (`PWR_CR5_R1MODE = 0`) and sets Flash latency to 5 wait states (`FLASH_ACR_LATENCY_5WS`).
   - Powers up HSE (8 MHz) and locks PLL at 180 MHz ($\text{PLLM}=2, \text{PLLN}=90, \text{PLLR}=2$).
2. **Peripheral Dividers**:
   - **USART1**: $\text{BRR} = \frac{180\,000\,000}{115200} \approx 1563$ ($115\,163\text{ baud}$, $+0.03\%$ error).
   - **I2C2**: Configured for 100 kHz standard mode using prescaler 17 ($t_{\text{PRESC}} = 100\text{ ns}$), $\text{SCLL} = 52$ ($5.3\ \mu\text{s}$), and $\text{SCLH} = 46$ ($4.7\ \mu\text{s}$). Open-drain mode allows safe operation on the sensor's $1.8\text{ V}$ bus rail.
   - **TIM3**: Counter clock prescaled by 180 ($\text{PSC}=179$) to 1 MHz. $\text{ARR}=999$ and $\text{CCR1}=500$ yield a 1 kHz, 50% duty cycle PWM on PB4.
3. **Hardware Pin Allocation Warning**:
   - Both Message 2 (USART1 TX) and Message 3 (I2C2 SCL) requested **pin PA9**. A GPIO pin can only be routed to one internal peripheral function at a time. In the combined code, PA9 is assigned to `AF4` (`I2C2_SCL`). To restore USART1 TX functionality, its TX signal must be mapped to an alternative pin (e.g., `PB6` AF7 or `PC4` AF7) [RM0440 p.304].
