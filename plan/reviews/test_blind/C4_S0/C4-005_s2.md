#include "stm32g4xx.h"

void Clock_Config_170MHz(void)
{
    /* 1. Enable PWR clock and switch regulator to Range 1 Boost Mode (needed for 170 MHz) [RM0440 p.232] */
    RCC->APB1ENR1 |= RCC_APB1ENR1_PWREN;
    PWR->CR5 &= ~PWR_CR5_R1MODE; // 0: Range 1 Boost mode enabled (f_HCLK up to 170 MHz)

    /* 2. Configure Flash Latency to 4 Wait States (WS) for 170 MHz in Range 1 Boost Mode [RM0440 p.119] */
    FLASH->ACR = (FLASH->ACR & ~FLASH_ACR_LATENCY_Msk) | FLASH_ACR_LATENCY_4WS;
    while ((FLASH->ACR & FLASH_ACR_LATENCY_Msk) != FLASH_ACR_LATENCY_4WS)
    {
        /* Wait for Flash latency update */
    }

    /* 3. Enable HSE oscillator (8 MHz external crystal) */
    RCC->CR |= RCC_CR_HSEON;
    while (!(RCC->CR & RCC_CR_HSERDY))
    {
        /* Wait until HSE is stable */
    }

    /* 4. Configure PLL:
     *    Source = HSE (8 MHz)
     *    PLLM   = 2  --> f_IN = 8 MHz / 2 = 4 MHz (PLLM bits: 0001 = /2)
     *    PLLN   = 85 --> f_VCO = 4 MHz * 85 = 340 MHz
     *    PLLR   = 2  --> f_PLL_R = 340 MHz / 2 = 170 MHz (PLLR bits: 00 = /2)
     *    Accuracy: exact (0.0% error)
     */
    RCC->CR &= ~RCC_CR_PLLON;
    while (RCC->CR & RCC_CR_PLLRDY)
    {
        /* Wait until PLL is shut down before reconfiguring */
    }

    RCC->PLLCFGR = RCC_PLLCFGR_PLLSRC_HSE |
                   (1U << RCC_PLLCFGR_PLLM_Pos) |       // PLLM = 2
                   (85U << RCC_PLLCFGR_PLLN_Pos) |      // PLLN = 85
                   RCC_PLLCFGR_PLLREN;                  // PLLR = 2 (00b) and enable PLL "R" output

    /* 5. Enable PLL and wait for lock */
    RCC->CR |= RCC_CR_PLLON;
    while (!(RCC->CR & RCC_CR_PLLRDY))
    {
        /* Wait until PLL is locked */
    }

    /* 6. Ensure AHB, APB1, and APB2 prescalers are set to /1 (reset default = 1) */
    RCC->CFGR &= ~(RCC_CFGR_HPRE | RCC_CFGR_PPRE1 | RCC_CFGR_PPRE2);

    /* 7. Switch SYSCLK source to PLL (PLLCLK) */
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW) | RCC_CFGR_SW_PLL;
    while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL)
    {
        /* Wait until PLL is selected as SYSCLK */
    }
}
```

### Explanation

1. **Power Range 1 Boost Mode**: STM32G474 requires Boost mode (`PWR_CR5_R1MODE = 0`) to operate beyond 150 MHz up to 170 MHz [RM0440 p.232].
2. **Flash Latency**: In Range 1 Boost mode, operating between 136 MHz and 170 MHz requires 4 wait states (`FLASH_ACR_LATENCY_4WS`) [RM0440 p.119].
3. **PLL Calculation**:
   $$\text{Input: } 8\text{ MHz (HSE)}$$
   $$f_{\text{VCO\_IN}} = \frac{8\text{ MHz}}{\text{PLLM (2)}} = 4\text{ MHz}$$
   $$f_{\text{VCO}} = 4\text{ MHz} \times \text{PLLN (85)} = 340\text{ MHz}$$
   $$f_{\text{PLL\_R}} = \frac{340\text{ MHz}}{\text{PLLR (2)}} = 170\text{ MHz}$$
4. **Prescalers**: Setting $\text{HPRE}=1$, $\text{PPRE1}=1$, and $\text{PPRE2}=1$ ensures $\text{SYSCLK} = \text{HCLK} = \text{PCLK1} = \text{PCLK2} = 170\text{ MHz}$ exactly (0.00% error).

{

     *    At 200 MHz, 5 Wait States (WS) are required (extrapolating 34 MHz/WS from Range 1 Boost).

    {
        /* Wait for Flash latency to update */
    }

    /* 3. Enable HSE (8 MHz external crystal) */
    RCC->CR |= RCC_CR_HSEON;
    while (!(RCC->CR & RCC_CR_HSERDY))
    {
        /* Wait for HSE ready */
    }

    /* 4. Configure PLL for 200 MHz:
     *    f_VCO_IN = 8 MHz / PLLM(2) = 4 MHz
     *    f_VCO    = 4 MHz * PLLN(100) = 400 MHz  (VCO range: 128 to 544 MHz)
     *    f_PLL_R  = 400 MHz / PLLR(2) = 200 MHz

    while (RCC->CR & RCC_CR_PLLRDY)
    {
        /* Wait until PLL is stopped */
    }

                   (1U << RCC_PLLCFGR_PLLM_Pos) |        // PLLM = 2
                   (100U << RCC_PLLCFGR_PLLN_Pos) |      // PLLN = 100

    /* 5. Enable PLL and wait for lock */
    RCC->CR |= RCC_CR_PLLON;
    while (!(RCC->CR & RCC_CR_PLLRDY))

        /* Wait for PLL lock */
    }

void TIM1_PA8_PWM_Init(void)
{
    /* 1. Enable peripheral clocks for GPIOA and TIM1 */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB2ENR |= RCC_APB2ENR_TIM1EN;

    /* 2. Configure PA8 as Alternate Function AF6 (TIM1_CH1) */
    GPIOA->MODER &= ~GPIO_MODER_MODE8_Msk;
    GPIOA->MODER |= (2U << GPIO_MODER_MODE8_Pos);        // AF mode (10b)

    GPIOA->AFR[1] &= ~GPIO_AFRH_AFSEL8_Msk;
    GPIOA->AFR[1] |= (6U << GPIO_AFRH_AFSEL8_Pos);       // AF6 = TIM1_CH1

    GPIOA->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED8_Pos);  // Very high speed for fast PWM edges
    GPIOA->PUPDR   &= ~GPIO_PUPDR_PUPD8_Msk;             // No pull-up / pull-down
    GPIOA->OTYPER  &= ~GPIO_OTYPER_OT8;                  // Push-pull

    /* 3. Configure TIM1 Time Base:
     *    f_TIM1_CLK = PCLK2 = 200 MHz (APB2 prescaler is 1)
     *    f_PWM      = 100 kHz
     *    Period     = 200 MHz / 100 kHz = 2000 counts
     *    ARR        = 2000 - 1 = 1999
     *    PSC        = 0 (no prescaling, highest resolution)
     */
    TIM1->PSC = 0;
    TIM1->ARR = 1999;

    /* 4. Configure Channel 1: PWM Mode 1, 50% Duty Cycle
     *    CCR1 = 2000 * 50% = 1000 counts
     */
    TIM1->CCR1 = 1000;

    // OC1M = 0110 (PWM Mode 1: active while CNT < CCR1), enable preload (OC1PE)
    TIM1->CCMR1 &= ~TIM_CCMR1_OC1M_Msk;
    TIM1->CCMR1 |= (6U << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE;

    // Active-high polarity (CC1P = 0) and enable output (CC1E = 1)
    TIM1->CCER &= ~TIM_CCER_CC1P;
    TIM1->CCER |= TIM_CCER_CC1E;

    /* 5. Advanced Timer Requirement: Enable Main Output (MOE) */
    TIM1->BDTR |= TIM1_BDTR_MOE;

    /* 6. Enable Auto-Reload Preload, generate an update event to latch ARR/CCR1, and start timer */
    TIM1->CR1 |= TIM_CR1_ARPE;
    TIM1->EGR  = TIM_EGR_UG;
    TIM1->CR1 |= TIM_CR1_CEN;
}

### Explanation

1. **200 MHz Clock**: By setting $\text{PLLN} = 100$ with $\text{PLLM} = 2$ and $\text{PLLR} = 2$, $f_{\text{VCO}} = 400\text{ MHz}$ and $f_{\text{SYSCLK}} = 200\text{ MHz}$. With all bus prescalers at $/1$, $f_{\text{HCLK}} = f_{\text{PCLK1}} = f_{\text{PCLK2}} = 200\text{ MHz}$.
2. **Duty-Cycle Resolution**: At $200\text{ MHz}$ clock and $100\text{ kHz}$ switching frequency, the counter spans $\text{ARR}+1 = 2000$ discrete steps, yielding an edge-aligned PWM resolution of $\frac{1}{2000} = 0.05\%$ ($11\text{ bits}$).
3. **GPIO Alternate Function**: PA8 is configured to Alternate Function 6 (`AF6`), which maps to `TIM1_CH1`.
4. **Main Output Enable (`MOE`)**: Advanced-control timers (TIM1 and TIM8) require bit `MOE` in `TIM1->BDTR` to be set to $1$; otherwise, the channel outputs remain disconnected from the pins.

#include "stm32g4xx.h"

int main(void)
{
    /* =========================================================================
     * 1. CLOCK CONFIGURATION (SYSCLK = HCLK = PCLK1 = PCLK2 = 200 MHz from 8 MHz HSE)
     * ========================================================================= */

    /* 1.1 Enable PWR clock and set regulator to Range 1 Boost Mode */
    RCC->APB1ENR1 |= RCC_APB1ENR1_PWREN;
    PWR->CR5 &= ~PWR_CR5_R1MODE; // 0: Boost Mode enabled

    /* 1.2 Set Flash Latency to 5 Wait States */
    FLASH->ACR = (FLASH->ACR & ~FLASH_ACR_LATENCY_Msk) | FLASH_ACR_LATENCY_5WS;
    while ((FLASH->ACR & FLASH_ACR_LATENCY_Msk) != FLASH_ACR_LATENCY_5WS)
    {
        /* Busy-wait for Flash latency update */
    }

    /* 1.3 Enable HSE oscillator (8 MHz external crystal) */
    RCC->CR |= RCC_CR_HSEON;
    while (!(RCC->CR & RCC_CR_HSERDY))
    {
        /* Busy-wait until HSE is ready */
    }

    /* 1.4 Disable PLL before configuring */
    RCC->CR &= ~RCC_CR_PLLON;
    while (RCC->CR & RCC_CR_PLLRDY)
    {
        /* Busy-wait until PLL is fully stopped */
    }

    /* 1.5 Configure PLL:
     *     Source = HSE (8 MHz)
     *     PLLM   = /2   (1U << PLLM_Pos)  --> 4 MHz VCO input
     *     PLLN   = *100 (100U << PLLN_Pos) --> 400 MHz VCO frequency
     *     PLLR   = /2   (00b)             --> 200 MHz PLLCLK output
     *     Enable PLL "R" output (PLLREN)
     */
    RCC->PLLCFGR = RCC_PLLCFGR_PLLSRC_HSE |
                   (1U << RCC_PLLCFGR_PLLM_Pos) |
                   (100U << RCC_PLLCFGR_PLLN_Pos) |
                   RCC_PLLCFGR_PLLREN;

    /* 1.6 Enable PLL and wait for lock */
    RCC->CR |= RCC_CR_PLLON;
    while (!(RCC->CR & RCC_CR_PLLRDY))
    {
        /* Busy-wait until PLL locks */
    }

    /* 1.7 Ensure AHB, APB1, and APB2 prescalers are set to /1 */
    RCC->CFGR &= ~(RCC_CFGR_HPRE | RCC_CFGR_PPRE1 | RCC_CFGR_PPRE2);

    /* 1.8 Switch SYSCLK source to PLLCLK */
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW) | RCC_CFGR_SW_PLL;
    while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL)
    {
        /* Busy-wait until PLL is selected as SYSCLK */
    }

    /* =========================================================================
     * 2. GPIO CONFIGURATION (PA8 -> TIM1_CH1, AF6)
     * ========================================================================= */

    /* 2.1 Enable peripheral clocks for GPIOA and TIM1 */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB2ENR |= RCC_APB2ENR_TIM1EN;

    /* 2.2 Configure PA8: AF mode, AF6 (TIM1_CH1), Very High Speed, Push-Pull, No-Pull */
    GPIOA->MODER   = (GPIOA->MODER & ~GPIO_MODER_MODE8_Msk) | (2U << GPIO_MODER_MODE8_Pos);
    GPIOA->AFR[1]  = (GPIOA->AFR[1] & ~GPIO_AFRH_AFSEL8_Msk) | (6U << GPIO_AFRH_AFSEL8_Pos);
    GPIOA->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED8_Pos);
    GPIOA->PUPDR   &= ~GPIO_PUPDR_PUPD8_Msk;
    GPIOA->OTYPER  &= ~GPIO_OTYPER_OT8;

    /* =========================================================================
     * 3. TIM1 CONFIGURATION (100 kHz PWM, 50% duty cycle, Edge-aligned)
     * ========================================================================= */

    /* 3.1 Prescaler and Period:
     *     Timer clock = 200 MHz
     *     PSC = 0 (divide by 1)
     *     ARR = 2000 - 1 = 1999  --> 200 MHz / 2000 = 100 kHz
     */
    TIM1->PSC = 0;
    TIM1->ARR = 1999;

    /* 3.2 Channel 1: 50% Duty Cycle (1000 / 2000) */
    TIM1->CCR1 = 1000;

    /* 3.3 Set PWM Mode 1 (OC1M = 0110b), enable channel 1 preload (OC1PE) */
    TIM1->CCMR1 = (TIM1->CCMR1 & ~TIM_CCMR1_OC1M_Msk) | (6U << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE;

    /* 3.4 Enable Channel 1 output with active-high polarity */
    TIM1->CCER &= ~TIM_CCER_CC1P;
    TIM1->CCER |= TIM_CCER_CC1E;

    /* 3.5 Enable Main Output (MOE) for advanced timer */
    TIM1->BDTR |= TIM_BDTR_MOE;

    /* 3.6 Enable ARR preload, trigger update to transfer shadow registers, and start counter */
    TIM1->CR1 |= TIM_CR1_ARPE;
    TIM1->EGR  = TIM_EGR_UG;
    TIM1->CR1 |= TIM_CR1_CEN;

    /* =========================================================================
     * 4. MAIN LOOP
     * ========================================================================= */
    while (1)
    {
        /* PWM runs continuously via hardware */
    }
}
```

### Explanation

1. **Clock Configuration**: 
   - The internal power regulator is set to **Range 1 Boost Mode** via `PWR->CR5`, and Flash latency is updated to 5 wait states to support 200 MHz operation.
   - The 8 MHz HSE crystal is divided by 2 ($\text{PLLM}=2 \rightarrow 4\text{ MHz}$), multiplied by 100 ($\text{PLLN}=100 \rightarrow 400\text{ MHz}$), and divided by 2 ($\text{PLLR}=2 \rightarrow 200\text{ MHz}$).
   - With all bus dividers ($\text{HPRE}, \text{PPRE1}, \text{PPRE2}$) left at $/1$, the entire core and peripheral domain runs at $200\text{ MHz}$.

2. **Pin Assignment**:
   - `PA8` is switched to alternate function mode `AF6` to connect the internal `TIM1_CH1` peripheral output directly to the pad.

3. **PWM Generation**:
   - With a $200\text{ MHz}$ input clock and prescaler set to $0$, $\text{ARR} = 1999$ produces an exact $100\text{ kHz}$ switching frequency ($200\text{ MHz} / 2000$).
   - $\text{CCR1} = 1000$ sets a $50\%$ duty cycle ($1000 / 2000$).
   - `TIM1_BDTR_MOE` enables the output driver circuitry specific to the advanced-control timer TIM1.
