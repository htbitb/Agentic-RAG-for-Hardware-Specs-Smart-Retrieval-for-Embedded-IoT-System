#include "stm32g4xx.h"

void usart2_init(void)
{
    /* 1. Enable GPIOA peripheral clock */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;

    /* 2. Configure PA2 (TX) and PA3 (RX) as Alternate Function 7 (AF7) */
    /* Clear MODER bits for PA2 and PA3 */
    GPIOA->MODER &= ~(GPIO_MODER_MODE2_Msk | GPIO_MODER_MODE3_Msk);
    /* Set MODER to 10 (Alternate Function mode) */
    GPIOA->MODER |= (2U << GPIO_MODER_MODE2_Pos) | (2U << GPIO_MODER_MODE3_Pos);

    /* Configure AF7 in AFRL for pins 2 and 3 */
    GPIOA->AFR[0] &= ~((0xFU << GPIO_AFRL_AFSEL2_Pos) | (0xFU << GPIO_AFRL_AFSEL3_Pos));
    GPIOA->AFR[0] |= (7U << GPIO_AFRL_AFSEL2_Pos) | (7U << GPIO_AFRL_AFSEL3_Pos);

    /* 3. Enable USART2 peripheral clock (default clock source is PCLK1 = 170 MHz) */
    RCC->APB1ENR1 |= RCC_APB1ENR1_USART2EN;

    /* 4. Configure Baud Rate:
     * Clock source = PCLK1 = 170,000,000 Hz
     * Oversampling by 16 (OVER8 = 0)
     * USARTDIV = 170000000 / 115200 ≈ 1475.694
     * BRR = (170000000 + 115200 / 2) / 115200 = 1476 (0x05C4)
     */
    USART2->BRR = (170000000UL + (115200UL / 2UL)) / 115200UL;

    /* 5. Clear CR1 configuration: 8 data bits (M[1:0] = 00), parity control disabled (PCE = 0),
     * oversampling by 16 (OVER8 = 0), and enable Transmitter and Receiver */
    USART2->CR1 = USART_CR1_TE | USART_CR1_RE;

    /* CR2 defaults to 1 stop bit (STOP[1:0] = 00) at reset */

    /* 6. Enable USART2 */
    USART2->CR1 |= USART_CR1_UE;
}
```

### Explanation

1. **GPIO Setup**: Pins PA2 and PA3 default to analog mode on reset. We switch both to Alternate Function mode (`MODER = 0b10`) and map them via `AFR[0]` (AFRL) to AF7, which corresponds to `USART2_TX` and `USART2_RX`.
2. **Peripheral Clock**: USART2 is clocked via `RCC->APB1ENR1`. At reset, `RCC->CCIPR` selects PCLK1 (170 MHz) as the USART2 clock source.
3. **Baud Rate Calculation**: With oversampling by 16 (`OVER8 = 0`), the baud rate divider is $USARTDIV = \frac{f_{CK}}{\text{Baud}}$. Rounding $170\,000\,000 / 115\,200$ gives `1476` (`0x05C4`), which is directly written into `USART2->BRR`.
4. **Ordering & Control**: `USART2->BRR` and transmitter/receiver enable bits (`TE`, `RE`) are written before asserting the USART enable bit (`UE`) in `CR1`. Word length defaults to 8 bits, parity to none, and stop bits to 1.

void i2c1_init(void)
{
    /* 1. Enable GPIOB and I2C1 peripheral clocks */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOBEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_I2C1EN;

    /* RCC_CCIPR I2C1SEL defaults to 00b (PCLK1 = 170 MHz), left at reset value */

    /* 2. Configure PB6 (SCL) and PB7 (SDA):
     * - Alternate Function mode (MODER = 10b)
     * - Open-Drain output type (OTYPER = 1)
     * - High speed (OSPEEDR = 11b)
     * - Pull-up enabled (PUPDR = 01b)
     * - Alternate Function AF4 (AFRL6 = 4, AFRL7 = 4)
     */
    GPIOB->MODER &= ~(GPIO_MODER_MODE6_Msk | GPIO_MODER_MODE7_Msk);
    GPIOB->MODER |= (2U << GPIO_MODER_MODE6_Pos) | (2U << GPIO_MODER_MODE7_Pos);

    GPIOB->OTYPER |= GPIO_OTYPER_OT6 | GPIO_OTYPER_OT7;

    GPIOB->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED6_Pos) | (3U << GPIO_OSPEEDR_OSPEED7_Pos);

    GPIOB->PUPDR &= ~(GPIO_PUPDR_PUPD6_Msk | GPIO_PUPDR_PUPD7_Msk);
    GPIOB->PUPDR |= (1U << GPIO_PUPDR_PUPD6_Pos) | (1U << GPIO_PUPDR_PUPD7_Pos);

    GPIOB->AFR[0] &= ~((0xFU << GPIO_AFRL_AFSEL6_Pos) | (0xFU << GPIO_AFRL_AFSEL7_Pos));
    GPIOB->AFR[0] |= (4U << GPIO_AFRL_AFSEL6_Pos) | (4U << GPIO_AFRL_AFSEL7_Pos);

    /* 3. Ensure I2C1 is disabled before configuring TIMINGR (PE = 0) */

    /* 4. Configure I2C_TIMINGR for Fm (400 kHz) with fI2CCLK = 170 MHz (tI2CCLK ≈ 5.882 ns):
     * - PRESC = 3  -> tPRESC = (3 + 1) * tI2CCLK = 4 * 5.882 ns ≈ 23.529 ns
     * - SCLL  = 64 -> tSCLL = (64 + 1) * 23.529 ns = 1529.4 ns
     *                 tLOW  = tSCLL + tSYNC1 = 1529.4 ns + 11.76 ns = 1541.2 ns (>= 1.3 µs)
     * - SCLH  = 39 -> tSCLH = (39 + 1) * 23.529 ns = 941.2 ns
     *                 tHIGH = tSCLH + tSYNC2 = 941.2 ns + 11.76 ns = 952.9 ns  (>= 0.6 µs)
     * - Total period tSCL = tSYNC1 + tSYNC2 + (SCLH + 1 + SCLL + 1) * tPRESC
     *                     = 4 * tI2CCLK + 105 * 4 * tI2CCLK = 424 * tI2CCLK ≈ 2494.1 ns
     *   -> fSCL ≈ 400.94 kHz (+0.24% of 400 kHz, within ±5%)
     * - SCLDEL = 4 -> tSCLDEL = (4 + 1) * 23.529 ns = 117.6 ns (tSU;DAT >= 100 ns)
     * - SDADEL = 0 -> tSDADEL = 0 ns (tHD;DAT >= 0 ns)
     */
    I2C1->TIMINGR = (3U << I2C_TIMINGR_PRESC_Pos)   |
                    (4U << I2C_TIMINGR_SCLDEL_Pos)  |
                    (0U << I2C_TIMINGR_SDADEL_Pos)  |
                    (39U << I2C_TIMINGR_SCLH_Pos)   |
                    (64U << I2C_TIMINGR_SCLL_Pos);

    /* 5. Enable I2C1 peripheral */
    I2C1->CR1 |= I2C_CR1_PE;
}
```

### Explanation

1. **GPIO Setup**: PB6 and PB7 are configured for Alternate Function 4 (AF4) as open-drain outputs with internal pull-ups enabled (complementing external pull-up resistors on the bus).
2. **Clock Source**: `RCC->CCIPR` retains its reset value of `00b` for `I2C1SEL`, selecting PCLK1 (170 MHz, $t_{I2CCLK} \approx 5.882\text{ ns}$).
3. **Timing Calculations** [RM0440 §39.4.9, Table 358]:
   - With $PRESC = 3$, prescaled clock cycle $t_{PRESC} = 4 \times t_{I2CCLK} \approx 23.529\text{ ns}$.

With $t_{SYNC1} = t_{SYNC2} = 2 \times t_{I2CCLK} \approx 11.76\text{ ns}$, setting $SCLL = 64$ gives $t_{LOW} \approx 1.541\ \mu\text{s}$ and setting $SCLH = 39$ gives $t_{HIGH} \approx 0.953\ \mu\text{s}$, meeting both requirements.

   - $SCLDEL = 4$ provides $t_{SU;DAT} = 117.6\text{ ns} \ge 100\text{ ns}$.
4. **Peripheral Enable**: The timing register is configured while $PE = 0$, followed by setting $PE = 1$ in `I2C1->CR1`.

### Explanation

1. **Clocks**: The peripheral clocks for `GPIOA`, `GPIOB`, `USART1`, `USART2`, and `I2C1` are enabled via their respective bus registers (`AHB2ENR`, `APB1ENR1`, `APB2ENR`).
2. **USART2**: Initialized on PA2/PA3 using AF7 at 115200 baud ($BRR = 1476$).
3. **I2C1**: Configured for 400 kHz Fast-mode ($PRESC=3, SCLL=64, SCLH=39, SCLDEL=4, SDADEL=0$) meeting $t_{LOW} \ge 1.3\ \mu\text{s}$ and $t_{HIGH} \ge 0.6\ \mu\text{s}$, and enabled via $PE=1$.
4. **USART1**: Configured at 921600 baud ($BRR = 184$, $+0.25\%$ error).
5. **Hardware Pin Collision on PB6/PB7**: Both I2C1 (PB6=SCL, PB7=SDA via AF4) and USART1 (PB6=TX, PB7=RX via AF7) were requested on the exact same physical pins. Because a GPIO pin can only be assigned to a single alternate function at a time (and I2C requires Open-Drain whereas UART TX uses Push-Pull), they cannot operate concurrently on PB6/PB7. In the code above, PB6 and PB7 are routed to USART1 (AF7). To operate both peripherals concurrently, one of them must be remapped to different physical pins (e.g., I2C1 on PA15/PB9 or PB8/PB9; USART1 on PA9/PA10).
