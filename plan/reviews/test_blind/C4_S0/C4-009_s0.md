#include "stm32g4xx.h"

void usart2_init(void)
{
    /* 1. Enable peripheral clocks for GPIOA and USART2 */
    RCC->AHB2ENR  |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_USART2EN;

    /* 2. Configure PA2 (TX) and PA3 (RX) for Alternate Function 7 (USART2) */
    /* Set mode to Alternate Function (10b) */
    GPIOA->MODER &= ~(GPIO_MODER_MODE2 | GPIO_MODER_MODE3);
    GPIOA->MODER |= (2U << GPIO_MODER_MODE2_Pos) | (2U << GPIO_MODER_MODE3_Pos);

    /* Set AF7 (0111b) in AFRL for pins 2 and 3 */
    GPIOA->AFR[0] &= ~((0xFU << GPIO_AFRL_AFSEL2_Pos) | (0xFU << GPIO_AFRL_AFSEL3_Pos));
    GPIOA->AFR[0] |=  (7U << GPIO_AFRL_AFSEL2_Pos)    | (7U << GPIO_AFRL_AFSEL3_Pos);

    /* Set high slew rate for high baud rate operation */
    GPIOA->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED2_Pos) | (3U << GPIO_OSPEEDR_OSPEED3_Pos);

    /* 3. Configure USART2 */
    /* Ensure USART2 is disabled before configuration */
    USART2->CR1 = 0;

    /* Frame format: 8 data bits (M[1:0] = 00), parity disabled (PCE = 0), 
       oversampling by 16 (OVER8 = 0) are defaults (0 in CR1) */
    /* 1 stop bit (STOP[1:0] = 00) is default (0 in CR2) */
    USART2->CR2 = 0;
    USART2->CR3 = 0;

    /* Baud rate calculation (PCLK1 = 170 MHz, OVER8 = 0):
       USARTDIV = 170,000,000 / 921,600 ≈ 184.4618
       BRR = round(184.4618) = 184
       Actual baud = 170,000,000 / 184 ≈ 923,913 baud (error: +0.25%, within ±2%) */
    USART2->BRR = 184U;

    /* Enable Transmitter and Receiver */
    USART2->CR1 |= USART_CR1_TE | USART_CR1_RE;

    /* Finally, enable USART2 */
    USART2->CR1 |= USART_CR1_UE;
}
```

### Explanation

* **Clock Source & Baud Rate:** By reset default in `RCC->CCIPR`, USART2 clock selection is configured to PCLK1 ($170\text{ MHz}$). With standard oversampling by 16 (`OVER8 = 0`), the division factor is $\text{BRR} = \frac{170\,000\,000}{921\,600} \approx 184$. The achieved baud rate is $\approx 923\,913\text{ baud}$, an error of $+0.25\%$, well within the $\pm 2\%$ limit.
* **GPIO Routing:** PA2 and PA3 are switched to alternate function mode (`MODER = 10b`) and mapped to AF7 (`USART2_TX` and `USART2_RX`) via `AFR[0]`. Slew rates are set to very high to ensure clean edges at 921.6 kbaud.
* **Control Registers:** `CR1` and `CR2` default to 8 data bits, no parity, and 1 stop bit. Per hardware requirements, `BRR`, `TE`, and `RE` are configured before setting the USART enable bit (`UE`).

    /* Configure PA2 as Open-Drain to avoid driving 3.3 V into the 1.8 V BLE RX pin */

    /* Ensure no internal pull-ups to 3.3 V are active on PA2 or PA3 */
    GPIOA->PUPDR &= ~(GPIO_PUPDR_PUPD2 | GPIO_PUPDR_PUPD3);

    /* Very high output speed */
    GPIOA->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED2_Pos) | (3U << GPIO_OSPEEDR_OSPEED3_Pos);

    /* 3. Configure USART2 */
    USART2->CR1 = 0;
    USART2->CR2 = 0;
    USART2->CR3 = 0;

    /* 921600 baud from 170 MHz PCLK1 */
    USART2->BRR = 184U;

    /* Enable TX, RX, and USART */
    USART2->CR1 |= USART_CR1_TE | USART_CR1_RE;
    USART2->CR1 |= USART_CR1_UE;

```

### Explanation

* **PA2 (TX $\rightarrow$ Module RX):** Configured as **Open-Drain** (`GPIOA->OTYPER |= GPIO_OTYPER_OT2`) with internal pull-ups disabled (`PUPDR = 0`). Driving 3.3 V push-pull exceeds the typical absolute maximum rating ($V_{DD} + 0.3\text{ V} \approx 2.1\text{ V}$) of a 1.8 V device. With open-drain, an external pull-up resistor to the module's 1.8 V supply rail safely defines the high level.
* **PA3 (Module TX $\rightarrow$ RX):** With $V_{DD} = 3.3\text{ V}$, the STM32G4 input high threshold is $V_{IH,\text{min}} = 0.47 \times V_{DD} + 0.25\text{ V} \approx 1.801\text{ V}$ [DS12288 p.76]. A 1.8 V output provides zero noise margin and will not reliably register a logic high. 
* **Hardware Requirement:** While open-drain protects the BLE RX, at **921600 baud** the RC rise time of a passive pull-up is critical ($t_{\text{bit}} \approx 1.085\,\mu\text{s}$). A dedicated dual-supply level translator IC (e.g., TXU0202 or 74LVC2T45) between the 3.3 V and 1.8 V domains is required for reliable bidirectional communication.
