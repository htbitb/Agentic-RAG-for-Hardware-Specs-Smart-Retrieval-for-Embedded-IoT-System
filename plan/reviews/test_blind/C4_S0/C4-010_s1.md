```c
#include "stm32g4xx.h"

void tim4_pwm_init(void)
{
    /* 1. Enable peripheral clocks for GPIOB and TIM4 */
    RCC->AHB2ENR  |= RCC_AHB2ENR_GPIOBEN;
    RCC->APB1ENR1 |= RCC_APB1ENR1_TIM4EN;

    /* 2. Configure PB6 and PB7 as Alternate Function AF2 (TIM4_CH1, TIM4_CH2) */
    /* Set mode to Alternate Function (10b) */
    GPIOB->MODER &= ~(GPIO_MODER_MODE6_Msk | GPIO_MODER_MODE7_Msk);
    GPIOB->MODER |=  (2U << GPIO_MODER_MODE6_Pos) | (2U << GPIO_MODER_MODE7_Pos);

    /* Set AF2 (0010b) in AFR[0] (AFRL) for pins 6 and 7 */
    GPIOB->AFR[0] &= ~((0xFU << GPIO_AFRL_AFSEL6_Pos) | (0xFU << GPIO_AFRL_AFSEL7_Pos));
    GPIOB->AFR[0] |=  (2U   << GPIO_AFRL_AFSEL6_Pos) | (2U   << GPIO_AFRL_AFSEL7_Pos);

    /* High speed output */
    GPIOB->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED6_Pos) | (3U << GPIO_OSPEEDR_OSPEED7_Pos);

    /* 3. Configure TIM4 time base for 10 kHz PWM */
    /* TIM4CLK = PCLK1 = 170 MHz (APB1 prescaler = 1) */
    /* Prescaler: 170 MHz / (16 + 1) = 10 MHz counter frequency */
    TIM4->PSC = 16U;

    /* Auto-reload: 10 MHz / 10 kHz = 1000 ticks -> ARR = 999 */
    TIM4->ARR = 999U;

    /* 4. Configure Duty Cycles */
    /* Channel 1: 30% of 1000 = 300 */
    TIM4->CCR1 = 300U;
    /* Channel 2: 60% of 1000 = 600 */
    TIM4->CCR2 = 600U;

    /* 5. Set PWM Mode 1 (OCxM = 0110b) and enable preload for CH1 and CH2 */
    TIM4->CCMR1 &= ~(TIM_CCMR1_OC1M_Msk | TIM_CCMR1_OC2M_Msk);
    TIM4->CCMR1 |=  (6U << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE;
    TIM4->CCMR1 |=  (6U << TIM_CCMR1_OC2M_Pos) | TIM_CCMR1_OC2PE;

    /* 6. Enable outputs, active-high polarity (CCxP = 0) */
    TIM4->CCER &= ~(TIM_CCER_CC1P | TIM_CCER_CC2P);
    TIM4->CCER |=  (TIM_CCER_CC1E | TIM_CCER_CC2E);

    /* 7. Enable ARR preload and generate an update event to latch PSC and ARR */
    TIM4->CR1 |= TIM_CR1_ARPE;
    TIM4->EGR  = TIM_EGR_UG;

    /* 8. Start TIM4 counter */
    TIM4->CR1 |= TIM_CR1_CEN;
}
```

### Explanation

1. **GPIO Setup**: PB6 and PB7 are mapped to AF2 (`TIM4_CH1` and `TIM4_CH2`) [DS12288 p.74].
2. **Frequency Derivation**: Since the APB1 prescaler is 1, the timer clock equals $f_{\text{PCLK1}} = 170\text{ MHz}$. Dividing by $\text{PSC} + 1 = 17$ yields a $10\text{ MHz}$ timer counter frequency. Setting $\text{ARR} = 999$ sets the period to $1000\text{ ticks}$ ($10\text{ kHz}$).
3. **PWM Generation**:
   * PWM Mode 1 (`OCxM = 0110b`) keeps the output active while $\text{CNT} < \text{CCRx}$ and inactive otherwise.
   * $\text{CCR1} = 300$ gives $\frac{300}{1000} = 30\%$ duty cycle.
   * $\text{CCR2} = 600$ gives $\frac{600}{1000} = 60\%$ duty cycle.
   * Generating an update event (`TIM_EGR_UG`) transfers the prescaler and preload values before setting `TIM_CR1_CEN`. TIM4 does not have a break/main output enable register (`BDTR`), so asserting `CCxE` is sufficient to drive the pins.

```c
void usart1_init(void)
{
    /* 1. Enable peripheral clocks for GPIOB and USART1 */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOBEN;
    RCC->APB2ENR |= RCC_APB2ENR_USART1EN;

    /* 2. Configure PB6 (TX) and PB7 (RX) to AF7 (USART1) */
    /* Set Alternate Function mode (10b) */
    GPIOB->MODER &= ~(GPIO_MODER_MODE6_Msk | GPIO_MODER_MODE7_Msk);
    GPIOB->MODER |=  (2U << GPIO_MODER_MODE6_Pos) | (2U << GPIO_MODER_MODE7_Pos);

    /* Set AF7 (0111b) in AFR[0] (AFRL) for pins 6 and 7 */
    GPIOB->AFR[0] &= ~((0xFU << GPIO_AFRL_AFSEL6_Pos) | (0xFU << GPIO_AFRL_AFSEL7_Pos));
    GPIOB->AFR[0] |=  (7U   << GPIO_AFRL_AFSEL6_Pos) | (7U   << GPIO_AFRL_AFSEL7_Pos);

    /* High speed output for PB6 (TX), pull-up on PB7 (RX) to prevent floating line */
    GPIOB->OSPEEDR |= (3U << GPIO_OSPEEDR_OSPEED6_Pos);
    GPIOB->PUPDR   &= ~GPIO_PUPDR_PUPD7_Msk;
    GPIOB->PUPDR   |=  (1U << GPIO_PUPDR_PUPD7_Pos);

    /* 3. Configure Baud Rate */
    /* USART1 clock source defaults to PCLK2 = 170 MHz (RCC_CCIPR.USART1SEL = 00b) */
    /* Oversampling by 16 (OVER8 = 0): BRR = fCK / baud = 170000000 / 115200 ≈ 1475.69 */
    /* Rounded to nearest integer: BRR = 1476 -> actual baud = 115176.15 baud (-0.02% error) */
    USART1->BRR = 1476U;

    /* 4. Configure frame format: 8 data bits (M1:M0 = 00), 1 stop bit (STOP = 00), no parity */
    USART1->CR2 &= ~USART_CR2_STOP;
    USART1->CR1 &= ~(USART_CR1_M1 | USART_CR1_M0 | USART_CR1_PCE | USART_CR1_OVER8);

    /* 5. Enable Transmitter and Receiver */
    USART1->CR1 |= (USART_CR1_TE | USART_CR1_RE);

    /* 6. Enable USART1 */
    USART1->CR1 |= USART_CR1_UE;
}

void usart1_write_char(char c)
{
    /* Wait until Transmit Data Register Empty (TXE / TXFNF) flag is set */
    while (!(USART1->ISR & USART_ISR_TXE_TXFNF))
    {
    }
    USART1->TDR = (uint8_t)c;
}

char usart1_read_char(void)
{
    /* Wait until Read Data Register Not Empty (RXNE / RXFNE) flag is set */
    while (!(USART1->ISR & USART_ISR_RXNE_RXFNE))
    {
    }
    return (char)(USART1->RDR & 0xFFU);
}
```

### Explanation

1. **GPIO Reconfiguration**: PB6 and PB7 are switched to **AF7** (`USART1_TX` and `USART1_RX`) [DS12288 p.74]. A pull-up is enabled on PB7 to keep the RX line idle-high when disconnected.
2. **Clock Source & Baud Rate**: 
   * By default, `RCC->CCIPR` selects `PCLK2` ($170\text{ MHz}$) for USART1.
   * With oversampling by 16 (`OVER8 = 0`), $\text{BRR} = \frac{f_{\text{CK}}}{\text{baud}} = \frac{170{,}000{,}000}{115{,}200} \approx 1475.69$.
   * Writing $\text{BRR} = 1476$ produces an actual baud rate of $115{,}176.15\text{ baud}$, yielding an error of approximately $-0.02\%$, which is well within the $\pm 2\%$ tolerance.
3. **Sequencing**:
   * Frame configuration (8 data bits, no parity, 1 stop bit) and `BRR` are written while the peripheral is disabled.
   * Transmitter (`TE`) and receiver (`RE`) bits are set, followed by enabling USART1 (`UE`).

### Explanation

1. **Hardware Pin Conflict**: PB6 and PB7 are physically shared between TIM4 (`AF2`: `TIM4_CH1`, `TIM4_CH2`) and USART1 (`AF7`: `USART1_TX`, `USART1_RX`) [DS12288 p.74]. A single GPIO pin can only be connected to one internal peripheral function via its multiplexer at any given time.
2. **Peripheral Engines**: Both TIM4 and USART1 are fully initialized and running simultaneously inside the MCU.
3. **GPIO Routing**: PB6 and PB7 are connected to **AF7** (USART1). If the board instead requires the TIM4 PWM outputs on those physical pins, assign `2U` to `AFSEL6` and `AFSEL7` in `GPIOB->AFR[0]`.
