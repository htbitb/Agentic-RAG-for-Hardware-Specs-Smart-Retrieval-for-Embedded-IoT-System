#include "stm32g4xx.h"

/**
 * @brief Configure clock tree:
 *        HSE (24 MHz Crystal) -> PLL -> SYSCLK = HCLK = PCLK1 = PCLK2 = 168 MHz
 *        AHB, APB1, APB2 prescalers = 1
 */
void SystemClock_Config(void)
{
    // 1. Enable PWR peripheral clock and configure Range 1 Boost mode (required for SYSCLK > 150 MHz)
    RCC->APB1ENR1 |= RCC_APB1ENR1_PWREN;
    PWR->CR5 &= ~PWR_CR5_R1MODE;

    // 2. Configure Flash memory to 4 wait states (required for 168 MHz in Range 1 Boost mode)
    FLASH->ACR = (FLASH->ACR & ~FLASH_ACR_LATENCY) | FLASH_ACR_LATENCY_4WS;

    // 3. Enable HSE oscillator and wait for it to be stable
    RCC->CR |= RCC_CR_HSEON;
    while (!(RCC->CR & RCC_CR_HSERDY)) {
        // Wait for HSE ready
    }

    // 4. Configure PLL:
    //    Source: HSE (24 MHz)
    //    PLLM: /2 (encoded as 1) -> fPLL_IN = 12 MHz (complies with datasheet fPLL_IN <= 16 MHz)
    //    PLLN: *28 -> VCO = 336 MHz
    //    PLLR: /2 (encoded as 00b) -> SYSCLK = 168 MHz
    //    Enable PLL 'R' clock output (PLLREN)
    RCC->PLLCFGR = RCC_PLLCFGR_PLLSRC_HSE |
                   (1U << RCC_PLLCFGR_PLLM_Pos) |
                   (28U << RCC_PLLCFGR_PLLN_Pos) |
                   RCC_PLLCFGR_PLLREN;

    // 5. Enable PLL and wait until locked
    RCC->CR |= RCC_CR_PLLON;
    while (!(RCC->CR & RCC_CR_PLLRDY)) {
        // Wait for PLL lock
    }

    // 6. Safe transition state for switching HCLK > 80 MHz (RM0440 p. 279-280):
    //    Set AHB prescaler to divide by 2 before switching
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_HPRE) | RCC_CFGR_HPRE_DIV2;

    // 7. Select PLL as system clock source and wait for switch confirmation
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_SW) | RCC_CFGR_SW_PLL;
    while ((RCC->CFGR & RCC_CFGR_SWS) != RCC_CFGR_SWS_PLL) {
        // Wait for clock switch
    }

    // 8. Wait at least 1 us, then restore AHB prescaler to divide by 1
    for (__IO uint32_t i = 0; i < 50; i++) {
        __NOP();
    }
    RCC->CFGR = (RCC->CFGR & ~RCC_CFGR_HPRE) | RCC_CFGR_HPRE_DIV1;
}

### Bare-Metal CMSIS C Code

```c
#include "stm32g4xx.h"

/**
 * @brief Initialize USART1 at 115200 baud, 8N1, Oversampling by 16.
 *        Pins: PA9 -> USART1_TX (AF7)
 *              PA10 -> USART1_RX (AF7)
 *        Kernel Clock: PCLK2 = 168 MHz (reset selection RCC_CCIPR.USART1SEL = 0)
 */
void USART1_Init(void)
{
    /* 1. Enable peripheral clock for GPIOA (AHB2) and USART1 (APB2) */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;
    RCC->APB2ENR |= RCC_APB2ENR_USART1EN;

    /* 2. Configure PA9 and PA10 alternate functions to AF7 (USART1_TX / USART1_RX)
     *    PA9 is AFR[1] (AFRH) bits [7:4]
     *    PA10 is AFR[1] (AFRH) bits [11:8]
     */
    GPIOA->AFR[1] = (GPIOA->AFR[1] & ~(GPIO_AFRH_AFSEL9_Msk | GPIO_AFRH_AFSEL10_Msk)) |
                    (7U << GPIO_AFRH_AFSEL9_Pos) |
                    (7U << GPIO_AFRH_AFSEL10_Pos);

    /* Set PA9 and PA10 mode to Alternate Function (10b) */
    GPIOA->MODER = (GPIOA->MODER & ~(GPIO_MODER_MODE9_Msk | GPIO_MODER_MODE10_Msk)) |
                   (2U << GPIO_MODER_MODE9_Pos) |
                   (2U << GPIO_MODER_MODE10_Pos);

    /* 3. Configure Baud Rate:
     *    With PCLK2 = 168 MHz and OVER8 = 0 (oversampling by 16):
     *    USARTDIV = 168000000 / 115200 = 1458.333...
     *    BRR = 1458 (0x05B2) -> Actual Baud = 115226.3 baud (+0.023% error)
     */
    USART1->BRR = 1458U;

    /* 4. Configure Stop bits: 1 Stop bit (STOP[1:0] = 00b in CR2) */
    USART1->CR2 &= ~USART_CR2_STOP;

    /* 5. Configure Word length, parity, oversampling, and enable TE, RE, UE:
     *    - M[1:0] = 00b: 8 data bits
     *    - PCE = 0: no parity
     *    - OVER8 = 0: oversampling by 16
     *    - TE = 1: transmitter enable
     *    - RE = 1: receiver enable
     *    - UE = 1: USART enable
     */
    USART1->CR1 = USART_CR1_TE | USART_CR1_RE | USART_CR1_UE;
}
```

### Architectural & Functional Explanation

1. **Core Voltage Range 1 Boost Mode**:
   - For frequencies exceeding 150 MHz up to 170 MHz, the STM32G474 internal voltage regulator must operate in **Range 1 Boost mode** [STM32G474_Datasheet p.123, STM32G474_RM0440 p.232].
   - Enabling `RCC_APB1ENR1_PWREN` [STM32G474_RM0440 p.309] and clearing `PWR_CR5_R1MODE` to `0` [STM32G474_RM0440 p.232, STM32G474_RM0440 p.267] places the device into Boost mode.

2. **Flash Wait States**:
   - According to RM0440 Table 9, at 168 MHz in Range 1 Boost mode (where HCLK $\le 170\text{ MHz}$), reading from Flash memory requires **4 wait states (5 CPU cycles)** (`FLASH_ACR_LATENCY_4WS`) [STM32G474_RM0440 p.98].

3. **PLL Multipliers & Electrical Limit Compliance**:
   - STM32G474 Datasheet Table 46 specifies that the PLL input frequency $f_{\text{PLL\_IN}}$ must strictly be between 2.66 MHz and 16 MHz [STM32G474_Datasheet p.123].
   - Supplying 24 MHz directly with $\text{PLLM}=1$ yields $f_{\text{PLL\_IN}} = 24\text{ MHz}$, exceeding the 16 MHz maximum limit.
   - Using $\text{PLLM} = 2$ scales HSE down to $24\text{ MHz} / 2 = 12\text{ MHz}$, which is compliant [STM32G474_Datasheet p.123].
   - Using $\text{PLLN} = 28$ produces a VCO frequency of $12\text{ MHz} \times 28 = 336\text{ MHz}$, which lies within the Range 1 VCO limits of 96 MHz to 344 MHz [STM32G474_Datasheet p.123].
   - Dividing the 336 MHz VCO by $\text{PLLR} = 2$ provides the desired $336\text{ MHz} / 2 = 168\text{ MHz}$ system clock [STM32G474_Datasheet p.123].

4. **Safe Clock Switching Sequence**:
   - As specified in Table 38 of RM0440, when switching to a higher system frequency in Range 1 Boost mode, the system clock is divided by 2 using the AHB prescaler (`RCC_CFGR_HPRE_DIV2`), the switch to PLL is made (`RCC_CFGR_SW_PLL`), and upon confirmation (`RCC_CFGR_SWS_PLL`), the AHB prescaler is restored to divide by 1 (`RCC_CFGR_HPRE_DIV1`) [STM32G474_RM0440 p.233].

5. **GPIO & USART1 Configuration**:
   - GPIO Port A and USART1 clocks are enabled in `RCC->AHB2ENR` [STM32G474_RM0440 p.309] and `RCC->APB2ENR` [STM32G474_RM0440 p.331].
   - PA9 and PA10 are configured as Alternate Function 7 (AF7) for `USART1_TX` and `USART1_RX` in `GPIOA->AFR[1]` and `GPIOA->MODER`.
   - USART1 uses its reset clock source selection (PCLK2 = 168 MHz) with 16x oversampling.
   - For 115200 baud, $\text{USARTDIV} = 168\,000\,000 / 115\,200 = 1458.333$.
   - Writing 1458 into `USART1->BRR` establishes an actual rate of $168\,000\,000 / 1458 = 115\,226.34\text{ baud}$ (+0.023% error).
   - Finally, `USART1->CR1` enables the transmitter (`TE`), receiver (`RE`), and the USART peripheral (`UE`) with 8 data bits and no parity.
