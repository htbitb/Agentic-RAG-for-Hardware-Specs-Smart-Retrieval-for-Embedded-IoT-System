#include "stm32g4xx.h"

void usart2_init(void) {
    /* 1. Enable clock to GPIOA */
    RCC->AHB2ENR |= RCC_AHB2ENR_GPIOAEN;

    /* 2. Configure PA2 and PA3 as Alternate Function (MODER = 10b) */
    GPIOA->MODER &= ~(GPIO_MODER_MODE2_Msk | GPIO_MODER_MODE3_Msk);
    GPIOA->MODER |= (2U << GPIO_MODER_MODE2_Pos) | (2U << GPIO_MODER_MODE3_Pos);

    /* 3. Map PA2 to AF7 (USART2_TX) and PA3 to AF7 (USART2_RX) in AFRL (AFR[0]) */
    GPIOA->AFR[0] &= ~(GPIO_AFRL_AFSEL2_Msk | GPIO_AFRL_AFSEL3_Msk);
    GPIOA->AFR[0] |= (7U << GPIO_AFRL_AFSEL2_Pos) | (7U << GPIO_AFRL_AFSEL3_Pos);

    /* 4. Enable clock to USART2 on APB1 */
    RCC->APB1ENR1 |= RCC_APB1ENR1_USART2EN;

    /* 5. Configure USART2 frame format:
     *    - 8 data bits: M1 = 0, M0 = 0
     *    - Parity control disabled: PCE = 0
     *    - Oversampling by 16: OVER8 = 0
     */
    USART2->CR1 &= ~(USART_CR1_M1 | USART_CR1_M0 | USART_CR1_PCE | USART_CR1_OVER8);

    /* 6. Configure 1 stop bit: STOP[1:0] = 00b in CR2 */
    USART2->CR2 &= ~USART_CR2_STOP_Msk;

    /* 7. Write baud rate register BEFORE enabling the USART
     *    Kernel clock = PCLK1 = 170 MHz, Target baud = 115200, OVER8 = 0
     *    BRR = 170000000 / 115200 = 1476 (0x05C4) -> 115176 baud (-0.02% error)
     */
    USART2->BRR = 1476U;

    /* 8. Enable Transmitter (TE), Receiver (RE), and USART (UE) */
    USART2->CR1 |= USART_CR1_TE | USART_CR1_RE | USART_CR1_UE;
}

     * - PA2 -> AF7 (USART2_TX)

    GPIOA->AFR[0] |= (7U << GPIO_AFRL_AFSEL2_Pos) | (7U << GPIO_AFRL_AFSEL3_Pos);

     * I2C1 GPIO Configuration:
     * - PB7 -> AF4 (I2C1_SDA), Open-Drain, internal pull-up, high speed
     * - PB8 -> AF4 (I2C1_SCL), Open-Drain, internal pull-up, high speed

    GPIOB->MODER |= (2U << GPIO_MODER_MODE7_Pos) | (2U << GPIO_MODER_MODE8_Pos);
